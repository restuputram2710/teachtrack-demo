function acak_(n, huruf) {
  huruf = huruf || 'ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnpqrstuvwxyz23456789';
  let s = '';
  while (s.length < n) {
    Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, Utilities.getUuid())
      .forEach(function (b) { if (s.length < n) s += huruf[(b & 0xff) % huruf.length]; });
  }
  return s;
}
function sandiAcak_(n) {
  const s = acak_(n || 10, 'ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnpqrstuvwxyz23456789');
  return /\d/.test(s) && /[A-Za-z]/.test(s) ? s : sandiAcak_(n);
}
function hashPass_(pass, salt) {
  let h = String(pass);
  for (let i = 0; i < 300; i++) h = Utilities.base64Encode(Utilities.computeHmacSha256Signature(h + salt, salt));
  return h;
}
function samaAman_(a, b) {
  a = String(a); b = String(b);
  if (a.length !== b.length) return false;
  let r = 0; for (let i = 0; i < a.length; i++) r |= a.charCodeAt(i) ^ b.charCodeAt(i);
  return r === 0;
}
function samarkan_(v) { v = String(v || ''); return !v ? '' : v.length <= 8 ? '••••••' : v.slice(0, 4) + '••••••' + v.slice(-6); }

/** Sandi pengguna disimpan sebagai PW_<id> = salt$hash di Script Properties (bukan di Sheets). */
function setSandiPengguna_(id, sandi) {
  const salt = Utilities.getUuid();
  setProp_('PW_' + id, salt + '$' + hashPass_(sandi, salt));
}
function cekSandiPengguna_(id, sandi) {
  const v = prop_('PW_' + id);
  if (!v || v.indexOf('$') < 0) return false;
  const salt = v.split('$')[0], hash = v.slice(salt.length + 1);
  return samaAman_(hashPass_(sandi, salt), hash);
}
function validasiSandiBaru_(s) {
  s = String(s || '');
  if (s.length < 8) return 'Kata sandi minimal 8 karakter.';
  if (!/[A-Za-z]/.test(s) || !/\d/.test(s)) return 'Kata sandi harus memuat huruf dan angka.';
  return '';
}

const POLA_RAHASIA = /(token|password|passwd|sandi|secret|rahasia|credential|kredensial|api[_-]?key|apikey|hash|salt|authorization|otp|pin|foto|lampiran|asli|thumb)/i;
function redaksi_(o, d) {
  d = d || 0;
  if (o === null || typeof o !== 'object' || d > 4) return o;
  if (Array.isArray(o)) return o.slice(0, 20).map(function (x) { return redaksi_(x, d + 1); });
  const out = {};
  Object.keys(o).forEach(function (k) { out[k] = POLA_RAHASIA.test(k) ? '[DISAMARKAN]' : redaksi_(o[k], d + 1); });
  return out;
}

function secret_() {
  let s = prop_('SESSION_SECRET');
  if (!s) { s = acak_(64); setProp_('SESSION_SECRET', s); }
  return s;
}
function hmac_(s) {
  return Utilities.base64EncodeWebSafe(Utilities.computeHmacSha256Signature(s, secret_(), Utilities.Charset.UTF_8)).replace(/=+$/, '');
}
function signToken_(p) {
  const body = Utilities.base64EncodeWebSafe(JSON.stringify(p), Utilities.Charset.UTF_8).replace(/=+$/, '');
  return 'h1.' + body + '.' + hmac_(body);
}
/** u = id pengguna / email (pending), r = peran token ('user' | 'pending' | 'darurat') */
function buatSesi_(u, r, ingat) {
  const now = Date.now();
  const umur = ingat ? SESI_INGAT_HARI * 864e5 : SESI_JAM * 3600e3;
  return signToken_({ u: u, r: r, i: now, x: now + umur });
}
function verifyToken_(t) {
  try {
    const parts = String(t || '').split('.');
    if (parts.length !== 3 || parts[0] !== 'h1') return null;
    if (!samaAman_(hmac_(parts[1]), parts[2])) return null;
    let b = parts[1]; while (b.length % 4) b += '=';
    const p = JSON.parse(Utilities.newBlob(Utilities.base64DecodeWebSafe(b)).getDataAsString());
    if (!p.x || p.x < Date.now()) return null;
    const rev = Number(CacheService.getScriptCache().get('rev_' + p.u) || prop_('REV_' + p.u) || 0);
    if (p.i < rev) return null;
    return p;
  } catch (e) { return null; }
}
function revokeAll_(u) {
  const t = String(Date.now());
  setProp_('REV_' + u, t);
  CacheService.getScriptCache().put('rev_' + u, t, 21600);
}

const CacheX = {
  SZ: 90000,
  c: function () { return CacheService.getScriptCache(); },
  get: function (key) {
    try {
      const c = this.c(), head = c.get(key);
      if (head == null) return null;
      if (head.indexOf('__chunks:') !== 0) return JSON.parse(head);
      const n = +head.slice(9), keys = [];
      for (let i = 0; i < n; i++) keys.push(key + '#' + i);
      const parts = c.getAll(keys); let s = '';
      for (let i = 0; i < n; i++) { if (parts[keys[i]] == null) return null; s += parts[keys[i]]; }
      return JSON.parse(s);
    } catch (e) { return null; }
  },
  put: function (key, val, ttl) {
    try {
      const s = JSON.stringify(val), c = this.c();
      if (s.length <= this.SZ) { c.put(key, s, ttl); return true; }
      const n = Math.ceil(s.length / this.SZ); if (n > 25) return false;
      const m = {}; for (let i = 0; i < n; i++) m[key + '#' + i] = s.substr(i * this.SZ, this.SZ);
      c.putAll(m, ttl); c.put(key, '__chunks:' + n, ttl);
      return true;
    } catch (e) { return false; }
  }
};
const _VER = {};
function vers_(scopes) {
  const c = CacheService.getScriptCache();
  const need = scopes.filter(function (s) { return _VER[s] === undefined; });
  if (need.length) {
    const got = c.getAll(need.map(function (s) { return 'ver_' + s; }));
    const baru = {};
    need.forEach(function (s) {
      let v = got['ver_' + s];
      if (!v) { v = String(Date.now()) + Math.floor(Math.random() * 1000); baru['ver_' + s] = v; }
      _VER[s] = v;
    });
    if (Object.keys(baru).length) c.putAll(baru, 21600);
  }
  return scopes.map(function (s) { return _VER[s]; });
}
function bump_(scope) {
  const v = String(Date.now()) + Math.floor(Math.random() * 1000);
  CacheService.getScriptCache().put('ver_' + scope, v, 21600);
  _VER[scope] = v;
  delete _TBL[scope];
  if (scope === 'set') delete _TBL.__set;
  if (scope === 'abs') { delete _TBL.__absRecent; delete _TBL.__absAll; }
}

function fmtD_(d, pola) { return Utilities.formatDate(d, TZ, pola); }
function normCell_(v, k) {
  if (v instanceof Date) {
    if (/^jam/.test(k)) return fmtD_(v, 'HH:mm');
    if (/tanggal|berlaku|^minggu$/.test(k)) return fmtD_(v, 'yyyy-MM-dd');
    return fmtD_(v, "yyyy-MM-dd'T'HH:mm:ss");
  }
  if (typeof v === 'number' && (k === 'nip_nuptk' || k === 'no_hp')) return waOrNip_(v, k);
  if (typeof v === 'string') return v.trim();
  return v;
}
function waOrNip_(v, k) {
  let s = String(v).trim();
  if (k === 'no_hp') { s = s.replace(/[^\d]/g, ''); if (/^62/.test(s)) s = '0' + s.slice(2); else if (/^8/.test(s)) s = '0' + s; }
  return s;
}

/** Baca sheet langsung (tanpa cache). Mengembalikan array objek. */
function readSheet_(name) {
  const sh = sheet_(name);
  const v = sh.getDataRange().getValues();
  const h = v[0].map(String);
  const rows = [];
  for (let r = 1; r < v.length; r++) {

    for (let i = 0; i < h.length; i++) if (h[i]) o[h[i]] = normCell_(v[r][i], h[i]);

  }

}

/** Cari baris berdasarkan kolom kunci (dibaca segar dari sheet). */
function findRow_(sh, h, keyField, keyVal) {
  const ci = h.indexOf(keyField);
  if (ci < 0) throw new Error('Kolom ' + keyField + ' tidak ada.');
  const n = sh.getLastRow() - 1;
  if (n < 1) return -1;
  const col = sh.getRange(2, ci + 1, n, 1).getValues();
  for (let i = 0; i < col.length; i++) if (String(col[i][0]).trim() === String(keyVal)) return i + 2;
  return -1;
}
/** Ubah sebagian kolom satu baris — hanya sel yang berubah yang ditulis (blok kontigu). */
function updateObj_(name, keyField, keyVal, patch) {
  const sh = sheet_(name), h = headerOf_(sh);
  const r = findRow_(sh, h, keyField, keyVal);
  if (r < 0) throw new Error('Data tidak ditemukan.');
  const row = sh.getRange(r, 1, 1, h.length).getValues()[0];
  let minC = h.length, maxC = -1;
  Object.keys(patch).forEach(function (k) {
    const i = h.indexOf(k); if (i < 0) return;
    row[i] = patch[k] === undefined || patch[k] === null ? '' : patch[k];
    minC = Math.min(minC, i); maxC = Math.max(maxC, i);
  });
  if (maxC >= 0) sh.getRange(r, minC + 1, 1, maxC - minC + 1).setValues([row.slice(minC, maxC + 1)]);
  if (SCOPE_OF[name]) bump_(SCOPE_OF[name]);
  const o = {}; h.forEach(function (k, i) { o[k] = normCell_(row[i], k); });
  return o;
}
function deleteObj_(name, keyField, keyVal) {
  const sh = sheet_(name), h = headerOf_(sh);
  const r = findRow_(sh, h, keyField, keyVal);
  if (r < 0) return false;
  sh.deleteRow(r);
  if (SCOPE_OF[name]) bump_(SCOPE_OF[name]);
  return true;
}

let _LOCK_DEPTH = 0;
function withLock_(fn) {
  if (_LOCK_DEPTH > 0) return fn();
  const l = LockService.getScriptLock();
  if (!l.tryLock(20000)) throw new Error('Server sedang sibuk. Silakan coba lagi beberapa detik lagi.');
  _LOCK_DEPTH++;
  try { return fn(); }
  finally { _LOCK_DEPTH--; l.releaseLock(); }
}

/** Bangun konteks sesi dari token. */
function buildCtx_(token) {
  const ctx = { me: null, pending: null, darurat: false };
  const p = verifyToken_(token);
  if (!p) return ctx;
  if (p.r === 'pending') { ctx.pending = { email: p.u }; return ctx; }
  if (p.r === 'darurat') {
    ctx.darurat = true;
    ctx.me = { id: 'darurat', nama: 'Akses Darurat', nip: '-', peran: 'admin', ikut_absen: false, darurat: true };
    return ctx;
  }
  const u = userById_(p.u);
  if (!u || u.status_akun !== 'aktif') return ctx;
  ctx.me = publicUser_(u);
  ctx.user = u;
  return ctx;
}
function allowed_(route, ctx) {
  switch (route.roles) {
    case 'pub': return true;
    case 'pending': return !!ctx.pending;
    case 'login': return !!ctx.me;
    case 'guru': return !!ctx.me && (ctx.me.ikut_absen === true);
    case 'admin': return !!ctx.me && ctx.me.peran === 'admin';
  }
  return false;
}
function runAction_(action, ctx, data) {
  const route = ROUTES[action];
  if (!route) return fail_('Aksi tidak dikenal: ' + action);
  if (!allowed_(route, ctx)) {
    if (!ctx.me && !ctx.pending && route.roles !== 'pub') return fail_('Sesi berakhir. Silakan masuk kembali.', { code: 'AUTH' });
    return fail_('Anda tidak memiliki akses untuk tindakan ini.', { code: 'FORBIDDEN' });
  }
  return route.fn(ctx, data || {});
}

function doPost(e) {
  const t0 = Date.now();
  let res, action = '';
  try {
    const body = JSON.parse((e && e.postData && e.postData.contents) || '{}');
    action = String(body.action || '');
    const ctx = buildCtx_(body.token);
    if (action === 'batch') res = apiBatch_(ctx, body.data || {});
    else res = runAction_(action, ctx, body.data || {});
  } catch (err) {
    console.error(action + ': ' + err.message);
    res = fail_(err.message || 'Terjadi kesalahan di server.');
  }
  res.ms = Date.now() - t0;
  res.v = APP_VERSION;
  if (res.ms > 6000) { try { audit_({ pelaku: 'sistem' }, 'LAMBAT', action, res.ms + ' ms'); } catch (x) { } }
  return jsonOut_(res);
}

/** GET: status, warmup (?w=pub|admin), atau aksi publik ?action=app.config */
function doGet(e) {
  const p = (e && e.parameter) || {};
  try {
    if (p.w) { const r = warmCaches_(p.w === 'admin' ? 'admin' : 'pub'); return jsonOut_({ success: true, warm: r, v: APP_VERSION }); }
    if (p.action === 'app.config') return jsonOut_(apiAppConfig_());
    return jsonOut_({ success: true, app: 'TeachTrack API', v: APP_VERSION, waktu: nowIso_(), siap: !!CONFIG.SPREADSHEET_ID });
  } catch (err) { return jsonOut_(fail_(err.message)); }
}

/** Batch: maks. 10 aksi BACA dalam satu eksekusi; hak akses dicek per aksi. */
function apiBatch_(ctx, d) {
  const calls = (d.calls || []).slice(0, 10), out = [];
  calls.forEach(function (c) {
    const r = ROUTES[c.action];
    if (!r || r.write) { out.push(fail_('Aksi tidak boleh di-batch: ' + c.action)); return; }
    try { out.push(runAction_(c.action, ctx, c.data || {})); }
    catch (err) { out.push(fail_(err.message)); }
  });
  return ok_({ results: out });
}

function apiLogin_(ctx, d) {
  const username = String(d.username || '').trim().toLowerCase();
  const password = String(d.password || '');
  if (!username || !password) return fail_('Isi NIP/username dan kata sandi.');
  const c = CacheService.getScriptCache(), kKunci = 'lk_' + username, kGagal = 'lg_' + username;
  const kunci = Number(c.get(kKunci) || 0);
  if (kunci > Date.now()) return fail_('Akun dikunci sementara karena terlalu banyak percobaan. Coba lagi ' + Math.ceil((kunci - Date.now()) / 60000) + ' menit lagi.', { code: 'LOCKED' });
  const u = users_().filter(function (x) {
    return String(x.nip_nuptk).toLowerCase() === username || String(x.id_pengguna).toLowerCase() === username;
  })[0];
  const benar = u && cekSandiPengguna_(u.id_pengguna, password);
  if (!benar) {
    const n = Number(c.get(kGagal) || 0) + 1;
    c.put(kGagal, String(n), 900);
    if (n >= 5) { c.put(kKunci, String(Date.now() + 15 * 60000), 900); c.remove(kGagal); }
    Utilities.sleep(400);
    return fail_('NIP/username atau kata sandi salah.' + (n >= 3 && n < 5 ? ' Sisa percobaan: ' + (5 - n) + '.' : n >= 5 ? ' Akun dikunci 15 menit.' : ''));
  }
  c.remove(kGagal);
  if (u.status_akun !== 'aktif') return fail_('Akun Anda ' + (u.status_akun || 'tidak aktif') + '. Hubungi admin.');
  audit_({ me: { id: u.id_pengguna, nama: u.nama_lengkap } }, 'LOGIN', 'Pengguna', 'NIP/username');
  return ok_({ token: buatSesi_(u.id_pengguna, 'user', !!d.ingat), me: publicUser_(u), set: publicSet_() });
}

function verifyGoogleIdToken_(idToken) {
  const cid = prop_('GOOGLE_CLIENT_ID');
  if (!cid) throw new Error('Login Google belum diaktifkan admin (GOOGLE_CLIENT_ID belum diisi).');
  const r = UrlFetchApp.fetch('https://oauth2.googleapis.com/tokeninfo?id_token=' + encodeURIComponent(idToken), { muteHttpExceptions: true });
  if (r.getResponseCode() !== 200) throw new Error('Token Google tidak valid atau kedaluwarsa.');
  const p = JSON.parse(r.getContentText());
  if (p.aud !== cid) throw new Error('Token Google bukan untuk aplikasi ini.');
  if (['accounts.google.com', 'https://accounts.google.com'].indexOf(p.iss) === -1) throw new Error('Penerbit token tidak dikenal.');
  if (String(p.email_verified) !== 'true') throw new Error('Email Google belum terverifikasi.');
  if (Number(p.exp) * 1000 < Date.now()) throw new Error('Token Google kedaluwarsa.');
  return { email: String(p.email).toLowerCase(), nama: p.name || '', foto: p.picture || '' };
}

function apiGoogle_(ctx, d) {
  const g = verifyGoogleIdToken_(String(d.credential || ''));
  return withLock_(function () {
    const u = readSheet_(SH.user).filter(function (x) { return String(x.email_google).toLowerCase() === g.email; })[0];
    if (u) {
      if (u.google_status === 'disetujui' && u.status_akun === 'aktif') {
        audit_({ me: { id: u.id_pengguna, nama: u.nama_lengkap } }, 'LOGIN', 'Pengguna', 'Google');
        return ok_({ token: buatSesi_(u.id_pengguna, 'user', !!d.ingat), me: publicUser_(u), set: publicSet_() });
      }
      if (u.google_status === 'ditolak' || u.status_akun === 'ditolak') return fail_('Akun Google ini ditolak admin. Hubungi admin sekolah.', { code: 'DITOLAK' });
      if (u.google_status !== 'menunggu') updateObj_(SH.user, 'id_pengguna', u.id_pengguna, { google_status: 'menunggu', diubah_pada: nowIso_() });
      return ok_({ pending: true, token: buatSesi_(g.email, 'pending', true), email: g.email, nama: u.nama_lengkap || g.nama, tertaut: true });
    }
    appendObj_(SH.user, {
      id_pengguna: newId_('GR'), nama_lengkap: g.nama || g.email, nip_nuptk: '', email_google: g.email,
      peran: 'guru', status_akun: 'menunggu', ikut_absen: 'ya', google_status: 'menunggu',
      dibuat_pada: nowIso_(), diubah_pada: nowIso_()
    });
    audit_({ pelaku: g.email }, 'DAFTAR_GOOGLE', 'Pengguna', 'Menunggu persetujuan');
    return ok_({ pending: true, token: buatSesi_(g.email, 'pending', true), email: g.email, nama: g.nama, tertaut: false });
  });
}

  const email = ctx.pending.email;
  const u = users_().filter(function (x) { return String(x.email_google).toLowerCase() === email; })[0];

  return ok_({ approved: false, email: email, nama: u ? u.nama_lengkap : '' });
}

function haversine_(lat1, lng1, lat2, lng2) {
  const R = 6371000, rad = Math.PI / 180;
  const dLat = (lat2 - lat1) * rad, dLng = (lng2 - lng1) * rad;
  const a = Math.sin(dLat / 2) * Math.sin(dLat / 2) + Math.cos(lat1 * rad) * Math.cos(lat2 * rad) * Math.sin(dLng / 2) * Math.sin(dLng / 2);
  return 2 * R * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
}

/** Terbilang bahasa Indonesia */
function terbilang_(n) {
  n = Math.floor(Math.abs(Number(n) || 0));
  const s = ['', 'satu', 'dua', 'tiga', 'empat', 'lima', 'enam', 'tujuh', 'delapan', 'sembilan', 'sepuluh', 'sebelas'];
  function t(x) {
    if (x < 12) return s[x];
    if (x < 20) return t(x - 10) + ' belas';
    if (x < 100) return t(Math.floor(x / 10)) + ' puluh' + (x % 10 ? ' ' + t(x % 10) : '');
    if (x < 200) return 'seratus' + (x - 100 ? ' ' + t(x - 100) : '');
    if (x < 1000) return t(Math.floor(x / 100)) + ' ratus' + (x % 100 ? ' ' + t(x % 100) : '');
    if (x < 2000) return 'seribu' + (x - 1000 ? ' ' + t(x - 1000) : '');
    if (x < 1e6) return t(Math.floor(x / 1000)) + ' ribu' + (x % 1000 ? ' ' + t(x % 1000) : '');
    if (x < 1e9) return t(Math.floor(x / 1e6)) + ' juta' + (x % 1e6 ? ' ' + t(x % 1e6) : '');
    if (x < 1e12) return t(Math.floor(x / 1e9)) + ' miliar' + (x % 1e9 ? ' ' + t(x % 1e9) : '');
    return t(Math.floor(x / 1e12)) + ' triliun' + (x % 1e12 ? ' ' + t(x % 1e12) : '');
  }
  if (n === 0) return 'Nol rupiah';
  const r = t(n).replace(/\s+/g, ' ').trim() + ' rupiah';
  return r.charAt(0).toUpperCase() + r.slice(1);
}

  const id = prop_(key);
  if (id) { try { return DriveApp.getFolderById(id); } catch (e) { } }
  const parent = DriveApp.getFolderById(prop_(parentKey || 'DRIVE_ROOT_FOLDER_ID'));
  const it = parent.getFoldersByName(nama);
  const f = it.hasNext() ? it.next() : parent.createFolder(nama);
  setProp_(key, f.getId());
  return f;
}
/** Simpan dataURL (base64) ke Drive. */
function simpanDataUrl_(dataUrl, folder, nama, maksByte) {
  const m = String(dataUrl || '').match(/^data:([\w\/+.-]+);base64,(.+)$/);
  if (!m) throw new Error('Format berkas tidak valid.');
  const mime = m[1];
  if (!/^(image\/(jpeg|png|webp)|application\/pdf)$/.test(mime)) throw new Error('Jenis berkas tidak didukung (gunakan JPG, PNG, WebP, atau PDF).');
  const bytes = Utilities.base64Decode(m[2]);
  if (bytes.length > (maksByte || 2.5e6)) throw new Error('Ukuran berkas terlalu besar (maks. ' + Math.round((maksByte || 2.5e6) / 1e6 * 10) / 10 + ' MB).');
  return folder.createFile(Utilities.newBlob(bytes, mime, nama));
}
function fileToDataUrl_(id) {
  const f = DriveApp.getFileById(id), b = f.getBlob();
  return 'data:' + b.getContentType() + ';base64,' + Utilities.base64Encode(b.getBytes());
}

/** "Minggu ke" dalam bulan (1–5) untuk sebuah tanggal: hari 1–7 = I, 8–14 = II, dst. */
function mingguKe_(tgl) { return Math.ceil(Number(String(tgl).slice(8, 10)) / 7); }
const ROMAWI = ['', 'I', 'II', 'III', 'IV', 'V'];
/** Daftar minggu ke (angka) dari teks "I,III" / "1,3"; kosong = setiap minggu. */
function daftarMingguKe_(v) {
  return String(v === undefined || v === null ? '' : v).split(/[,\s;]+/).map(function (x) {
    x = x.trim().toUpperCase(); if (!x) return 0;
    const r = ROMAWI.indexOf(x); return r > 0 ? r : (Number(x) >= 1 && Number(x) <= 5 ? Number(x) : 0);
  }).filter(Boolean);
}
/** Jadwal yang berlaku untuk tanggal tertentu (periode + "minggu ke" bila diisi). */
function jadwalBerlaku_(j, tgl, abaikanMinggu) {
  if (j.aktif !== '' && j.aktif !== undefined && !isYa_(j.aktif)) return false;
  if (j.berlaku_mulai && tgl < j.berlaku_mulai) return false;
  if (j.berlaku_sampai && tgl > j.berlaku_sampai) return false;
  if (!abaikanMinggu) { const mk = daftarMingguKe_(j.minggu_ke); if (mk.length && mk.indexOf(mingguKe_(tgl)) < 0) return false; }
  return true;
}

/**
 * Bangun indeks untuk perhitungan status harian (sekali baca, dipakai berulang).
 * opts.tkb: hanya hitung absen & jadwal di TKB tertentu.
 */
function buildIndex_(dari, sampai, opts) {
  opts = opts || {};
  const absBy = {}, izinBy = {}, jadBy = {};
  absFor_(dari).forEach(function (a) {
    if (a.tanggal < dari || a.tanggal > sampai) return;
    if (opts.tkb && a.id_tkb !== opts.tkb) return;
    if (!absBy[a.id_pengguna]) absBy[a.id_pengguna] = {};
    absBy[a.id_pengguna][a.tanggal] = a;
  });
  izinRows_().forEach(function (z) {
    if (z.status !== 'disetujui') return;
    if (z.tanggal_selesai < dari || z.tanggal_mulai > sampai) return;
    const a = z.tanggal_mulai > dari ? z.tanggal_mulai : dari;
    const b = z.tanggal_selesai < sampai ? z.tanggal_selesai : sampai;
    if (!izinBy[z.id_pengguna]) izinBy[z.id_pengguna] = {};
    dateRange_(a, b).forEach(function (t) { izinBy[z.id_pengguna][t] = z; });
  });
  jadwalRows_().forEach(function (j) {
    if (opts.tkb && j.id_tkb !== opts.tkb) return;
    if (!jadBy[j.id_pengguna]) jadBy[j.id_pengguna] = [];
    jadBy[j.id_pengguna].push(j);
  });
  Object.keys(jadBy).forEach(function (k) { jadBy[k].sort(function (a, b) { return menit_(a.jam_mulai) - menit_(b.jam_mulai); }); });
  const tj = todayJ_();
  return { absBy: absBy, izinBy: izinBy, jadBy: jadBy, today: tj.tgl, nowMenit: menit_(tj.jam) };
}
function jadwalHari_(idx, uid, tgl) {
  const hari = hariOf_(tgl);
  return (idx.jadBy[uid] || []).filter(function (j) { return j.hari === hari && jadwalBerlaku_(j, tgl); });
}
/**
 * Kode status harian:
 *  H = hadir tepat waktu, T = terlambat, L = di luar jadwal,
 *  S = sakit, I = izin/dinas/lainnya, A = alpa (termasuk hari ini bila semua sesi sudah lewat),
 *  B = belum absen (hari ini, sesi masih berjalan/akan datang), - = tidak ada jadwal
 */
function statusHari_(idx, uid, tgl) {
  const a = idx.absBy[uid] && idx.absBy[uid][tgl];
  const jad = jadwalHari_(idx, uid, tgl);
  if (a) return { kode: a.status === 'terlambat' ? 'T' : a.status === 'di_luar_jadwal' ? 'L' : 'H', absen: a, jadwal: jad };
  const z = idx.izinBy[uid] && idx.izinBy[uid][tgl];
  if (z) return { kode: z.jenis === 'sakit' ? 'S' : 'I', izin: z, jadwal: jad };
  if (jad.length && tgl < idx.today) return { kode: 'A', jadwal: jad };
  if (jad.length && tgl === idx.today) {
    // Hari ini: menjadi Alpa bila seluruh sesi hari ini sudah berakhir tanpa absen
    const akhir = Math.max.apply(null, jad.map(function (j) { return menit_(j.jam_selesai); }));
    return { kode: idx.nowMenit > akhir ? 'A' : 'B', jadwal: jad };
  }
  return { kode: '-', jadwal: jad };
}
/** Ringkasan untuk satu guru dalam rentang tanggal. */
function ringkas_(idx, uid, dates, tarif) {
  const r = { H: 0, T: 0, L: 0, S: 0, I: 0, A: 0, B: 0, hadir: 0, terverifikasi: 0, menitTerlambat: 0, hariJadwal: 0 };
  dates.forEach(function (t) {
    const s = statusHari_(idx, uid, t);
    if (s.kode !== '-') r[s.kode]++;
    if (s.jadwal.length) r.hariJadwal++;
    if (s.absen) {
      r.hadir++;
      if (isYa_(s.absen.verifikasi)) r.terverifikasi++;
      r.menitTerlambat += num_(s.absen.menit_terlambat, 0);
    }
  });
  r.honor = r.terverifikasi * tarif;
  const dasar = r.hadir + r.A;
  r.persen = dasar ? Math.round(r.hadir / dasar * 1000) / 10 : 100;
  return r;
}

function apiGuruBoot_(ctx) {
  const uid = ctx.me.id, now = todayJ_(), set = getSet_();
  const tarif = num_(set.tarif_per_kehadiran, 0);
  const bulan = now.tgl.slice(0, 7);
  const dari = bulan + '-01';
  const idx = buildIndex_(addDays_(now.tgl, -40) < dari ? addDays_(now.tgl, -40) : dari, now.tgl);
  const datesBulan = dateRange_(dari, now.tgl);
  const sHari = statusHari_(idx, uid, now.tgl);

  const myJadwal = jadwalRows_().filter(function (j) { return j.id_pengguna === uid && jadwalBerlaku_(j, now.tgl, true); }).map(jadwalPublik_);
  myJadwal.sort(function (a, b) { return HARI.indexOf(a.hari) - HARI.indexOf(b.hari) || menit_(a.mulai) - menit_(b.mulai); });
  const tkbIds = {}; myJadwal.forEach(function (j) { tkbIds[j.id_tkb] = 1; });
  if (sHari.absen) tkbIds[sHari.absen.id_tkb] = 1;

  const agenda = agendaUntuk_(uid).map(agendaPublik_).filter(function (g) { return g.selesai >= addDays_(now.tgl, -30); })
    .sort(function (a, b) { return a.mulai < b.mulai ? -1 : a.mulai > b.mulai ? 1 : 0; });

  const izin = izinRows_().filter(function (z) { return z.id_pengguna === uid; }).map(izinPublik_)
    .sort(function (a, b) { return a.diajukan < b.diajukan ? 1 : -1; }).slice(0, 30);

  const riw = datesBulan.slice().reverse().map(function (t) { return harianPublik_(statusHari_(idx, uid, t), t); })
    .filter(function (x) { return x.kode !== '-'; });

  return ok_({
    me: ctx.me, set: publicSet_(),
    hariIni: { tanggal: now.tgl, hari: now.hari, jam: now.detik, kode: sHari.kode,
      absen: absPublik_(sHari.absen), izin: izinPublik_(sHari.izin), jadwal: sHari.jadwal.map(jadwalPublik_) },
    jadwal: myJadwal,
    tkb: Object.keys(tkbIds).map(function (id) { return tkbPublik_(tkbById_(id)); }).filter(Boolean),
    agenda: agenda,
    izin: izin,
    bulan: { kode: bulan, ringkasan: ringkas_(idx, uid, datesBulan, tarif), riwayat: riw },
    tarif: tarif
  });
}

    jadwal: s.jadwal.map(function (j) { return { mulai: j.jam_mulai, selesai: j.jam_selesai, mapel: j.mata_pelajaran, tkb: (tkbById_(j.id_tkb) || {}).nama_tkb || j.id_tkb }; }) };
}

/** Riwayat & honor per bulan (yyyy-MM) untuk guru yang login. */
function apiGuruRiwayat_(ctx, d) {
  const uid = ctx.me.id, now = todayJ_();
  const bulan = /^\d{4}-\d{2}$/.test(String(d.bulan || '')) ? d.bulan : now.tgl.slice(0, 7);
  const dari = bulan + '-01';
  const akhir = addDays_(addDays_(dari, 32).slice(0, 7) + '-01', -1);
  if (dari > now.tgl) return ok_({ bulan: bulan, ringkasan: null, riwayat: [], mingguan: [] });
  const sampai = akhir < now.tgl ? akhir : now.tgl;
  const tarif = num_(getSet_().tarif_per_kehadiran, 0);
  const idx = buildIndex_(dari, sampai);
  const dates = dateRange_(dari, sampai);
  const riwayat = dates.slice().reverse().map(function (t) { return harianPublik_(statusHari_(idx, uid, t), t); }).filter(function (x) { return x.kode !== '-'; });
  // Rincian mingguan (Senin–Minggu) untuk halaman Honor
  const minggu = [];
  dates.forEach(function (t) {
    const h = hariOf_(t);
    if (!minggu.length || h === 'Senin') minggu.push({ dari: t, sampai: t, hadir: 0, terverifikasi: 0, izin: 0, alpa: 0 });
    const w = minggu[minggu.length - 1]; w.sampai = t;
    const s = statusHari_(idx, uid, t);
    if (s.absen) { w.hadir++; if (isYa_(s.absen.verifikasi)) w.terverifikasi++; }
    if (s.kode === 'I' || s.kode === 'S') w.izin++;
    if (s.kode === 'A') w.alpa++;
  });
  minggu.forEach(function (w, i) { w.ke = i + 1; w.honor = w.terverifikasi * tarif; });
  return ok_({ bulan: bulan, ringkasan: ringkas_(idx, uid, dates, tarif), riwayat: riwayat, mingguan: minggu.reverse(), tarif: tarif, tutup: akhir });
}

function parseKode_(k) {
  k = String(k || '').trim();
  const m = k.match(/(?:TEACHTRACK:|[?&#]t=)([A-Za-z0-9-]+)/i);
  return (m ? m[1] : k).replace(/\s+/g, '').toUpperCase();
}
function tkbByToken_(kode) {
  const k = parseKode_(kode);
  if (!k) return { err: fail_('Kode QR kosong.', { code: 'QR_INVALID' }) };
  const t = tkbs_().filter(function (x) { return String(x.token_qr).toUpperCase() === k; })[0];
  if (!t) return { err: fail_('QR tidak berlaku. Pastikan Anda memindai QR resmi yang ditempel di TKB, atau minta QR terbaru dari admin.', { code: 'QR_INVALID' }) };
  if (t.status !== 'aktif') return { err: fail_('TKB ' + t.nama_tkb + ' sedang nonaktif. Hubungi admin.', { code: 'TKB_NONAKTIF' }) };
  if (!isFinite(Number(t.latitude)) || !isFinite(Number(t.longitude)) || t.latitude === '' || t.longitude === '')
    return { err: fail_('Titik lokasi TKB ' + t.nama_tkb + ' belum diatur admin.', { code: 'TKB_TANPA_LOKASI' }) };
  return { tkb: t };
}
/** Cocokkan jadwal: jadwal pertama yang belum lewat di TKB tersebut pada hari ini. */
function cocokJadwal_(uid, tkbId, tgl, jam, toleransi) {
  const hari = hariOf_(tgl), m = menit_(jam);
  const semua = jadwalRows_().filter(function (j) { return j.id_pengguna === uid && j.hari === hari && jadwalBerlaku_(j, tgl); })
    .sort(function (a, b) { return menit_(a.jam_mulai) - menit_(b.jam_mulai); });
  const diTkb = semua.filter(function (j) { return j.id_tkb === tkbId; });
  if (!diTkb.length) return { status: 'di_luar_jadwal', menit: 0, jadwal: null, jadwalLain: semua.map(jadwalPublik_) };
  let pilih = diTkb.filter(function (j) { return menit_(j.jam_selesai) >= m; })[0] || diTkb[diTkb.length - 1];
  const telat = m - menit_(pilih.jam_mulai);
  const terlambat = telat > toleransi;
  return { status: terlambat ? 'terlambat' : 'tepat_waktu', menit: terlambat ? telat : 0, jadwal: jadwalPublik_(pilih),
    batas: menitKeJam_(menit_(pilih.jam_mulai) + toleransi) };
}
function menitKeJam_(m) { return ('0' + Math.floor(m / 60)).slice(-2) + ':' + ('0' + (m % 60)).slice(-2); }

/** Cek cepat apakah guru sudah absen di tanggal ini (cache penanda + ekor sheet). */
function sudahAbsen_(uid, tgl, segar) {
  const c = CacheService.getScriptCache().get('absday_' + uid + '_' + tgl);
  if (c) return JSON.parse(c);
  if (!segar) {
    const r = absRecent_().filter(function (a) { return a.id_pengguna === uid && a.tanggal === tgl; })[0];
    return r || null;
  }
  const sh = sheet_(SH.abs), h = headerOf_(sh), last = sh.getLastRow();
  if (last < 2) return null;
  const from = Math.max(2, last - 3000);
  const v = sh.getRange(from, 1, last - from + 1, h.length).getValues();
  const iU = h.indexOf('id_pengguna'), iT = h.indexOf('tanggal');
  for (let i = v.length - 1; i >= 0; i--) {
    if (String(v[i][iU]) === uid && normCell_(v[i][iT], 'tanggal') === tgl) {
      const o = {}; h.forEach(function (k, j) { o[k] = normCell_(v[i][j], k); }); return o;
    }
  }
  return null;
}
function izinPada_(uid, tgl) {
  return izinRows_().filter(function (z) { return z.id_pengguna === uid && z.status === 'disetujui' && z.tanggal_mulai <= tgl && z.tanggal_selesai >= tgl; })[0] || null;
}

function apiAbsenCek_(ctx, d) {
  const uid = ctx.me.id, now = todayJ_(), set = getSet_();
  const sudah = sudahAbsen_(uid, now.tgl, false);
  if (sudah) return fail_('Anda sudah absen hari ini pukul ' + sudah.jam_absen + ' di ' + ((tkbById_(sudah.id_tkb) || {}).nama_tkb || sudah.id_tkb) + '.', { code: 'SUDAH_ABSEN', data: absPublik_(sudah) });
  const iz = izinPada_(uid, now.tgl);
  if (iz) return fail_('Hari ini Anda tercatat ' + (iz.jenis === 'sakit' ? 'sakit' : 'izin') + ' (sudah disetujui). Absen tidak diperlukan. Bila ternyata hadir, minta admin membatalkan izinnya.', { code: 'SEDANG_IZIN' });
  const r = tkbByToken_(d.kode);
  if (r.err) return r.err;
  const t = r.tkb, tp = tkbPublik_(t);
  const out = { tkb: tp, jamServer: now.detik, tanggal: now.tgl, hari: now.hari,
    cocok: cocokJadwal_(uid, t.id_tkb, now.tgl, now.jam, num_(set.toleransi_terlambat_menit, 15)),
    akurasiMaks: num_(set.akurasi_maks_meter, 100) };
  if (d.lat !== undefined && d.lat !== null && d.lat !== '') {
    const lat = Number(d.lat), lng = Number(d.lng), acc = Number(d.acc);
    if (!isFinite(lat) || !isFinite(lng)) return fail_('Koordinat GPS tidak valid.', { code: 'GPS' });
    out.jarak = Math.round(haversine_(lat, lng, tp.lat, tp.lng) * 10) / 10;
    out.akurasi = isFinite(acc) ? Math.round(acc) : null;
    out.dalamRadius = out.jarak <= tp.radius;
    out.akurasiOk = out.akurasi === null || out.akurasi <= out.akurasiMaks;
  }
  return ok_(out);
}

function apiAbsenKirim_(ctx, d) {
  const uid = ctx.me.id, set = getSet_();
  const r = tkbByToken_(d.kode);
  if (r.err) return r.err;
  const t = r.tkb, tp = tkbPublik_(t);
  const lat = Number(d.lat), lng = Number(d.lng), acc = Number(d.acc);
  if (!isFinite(lat) || !isFinite(lng)) return fail_('Lokasi GPS belum didapat. Aktifkan lokasi lalu coba lagi.', { code: 'GPS' });
  const maksAcc = num_(set.akurasi_maks_meter, 100);
  if (isFinite(acc) && acc > maksAcc) return fail_('Akurasi GPS terlalu rendah (±' + Math.round(acc) + ' m). Tunggu sinyal membaik lalu coba lagi.', { code: 'AKURASI' });
  const jarak = haversine_(lat, lng, tp.lat, tp.lng);
  if (jarak > tp.radius) return fail_('Anda berada ±' + Math.round(jarak) + ' m dari TKB ' + t.nama_tkb + '. Mendekatlah ke lokasi (radius ' + tp.radius + ' m).', { code: 'LUAR_RADIUS', jarak: Math.round(jarak) });
  if (!d.foto) return fail_('Foto bukti kehadiran wajib diambil.', { code: 'FOTO' });
  const jenis = d.jenis_foto === 'kegiatan' ? 'kegiatan' : 'selfie';

  const now0 = todayJ_();
  if (sudahAbsen_(uid, now0.tgl, false)) return fail_('Anda sudah absen hari ini.', { code: 'SUDAH_ABSEN' });
  if (izinPada_(uid, now0.tgl)) return fail_('Hari ini Anda tercatat izin/sakit.', { code: 'SEDANG_IZIN' });

  // Simpan foto DI LUAR lock (operasi terberat), lalu tulis baris di dalam lock singkat.
  const ym = now0.tgl.slice(0, 7);
  folderProp_('FOLDER_FOTO', 'DRIVE_ROOT_FOLDER_ID', 'Foto Absensi');
  const folder = folderProp_('FOLDER_FOTO_' + ym.replace('-', ''), 'FOLDER_FOTO', ym);
  const file = simpanDataUrl_(d.foto, folder, now0.tgl + '_' + uid + '_' + jenis + '.jpg', 3e6);

  try {
    const rec = withLock_(function () {
      const now = todayJ_();
      delete _VER.abs; delete _TBL.__absRecent;
      if (sudahAbsen_(uid, now.tgl, true)) return null;
      const cocok = cocokJadwal_(uid, t.id_tkb, now.tgl, now.jam, num_(set.toleransi_terlambat_menit, 15));
      const o = {
        id_absen: newId_('ABS'), tanggal: now.tgl, id_pengguna: uid, id_tkb: t.id_tkb, jam_absen: now.detik,
        latitude: lat, longitude: lng, akurasi_gps_meter: isFinite(acc) ? Math.round(acc) : '',
        jarak_ke_tkb_meter: Math.round(jarak * 10) / 10, foto_file_id: file.getId(), jenis_foto: jenis,
        id_jadwal_terkait: cocok.jadwal ? cocok.jadwal.id : '', status: cocok.status, menit_terlambat: cocok.menit,
        verifikasi: cocok.status === 'di_luar_jadwal' ? 'tidak' : 'ya', ditandai: '', diubah_admin: '',
        catatan_admin: '', perangkat: String(d.perangkat || '').slice(0, 120), dibuat_pada: nowIso_()
      };
      appendObj_(SH.abs, o);
      CacheService.getScriptCache().put('absday_' + uid + '_' + now.tgl, JSON.stringify(o), 86400);
      return { o: o, cocok: cocok };
    });
    if (!rec) { try { file.setTrashed(true); } catch (e) { } return fail_('Anda sudah absen hari ini.', { code: 'SUDAH_ABSEN' }); }
    return ok_({ absen: absPublik_(rec.o), cocok: rec.cocok, tkb: tp }, 'Absen berhasil dicatat.');
  } catch (err) {
    try { file.setTrashed(true); } catch (e) { }
    throw err;
  }
}

const JENIS_IZIN = ['izin', 'sakit', 'dinas', 'lainnya'];
function apiIzinAjukan_(ctx, d) {
  const uid = ctx.me.id, now = todayJ_();
  const jenis = String(d.jenis || '');
  if (JENIS_IZIN.indexOf(jenis) < 0) return fail_('Pilih kategori ketidakhadiran.');
  const mulai = String(d.mulai || ''), selesai = String(d.selesai || d.mulai || '');
  if (!validTgl_(mulai) || !validTgl_(selesai)) return fail_('Tanggal tidak valid.');
  if (selesai < mulai) return fail_('Tanggal selesai tidak boleh sebelum tanggal mulai.');
  if (dateRange_(mulai, selesai).length > 31) return fail_('Rentang izin maksimal 31 hari. Ajukan terpisah bila lebih.');
  if (mulai < addDays_(now.tgl, -7)) return fail_('Pengajuan mundur maksimal 7 hari ke belakang.');

  const bentrok = izinRows_().filter(function (z) {
    return z.id_pengguna === uid && (z.status === 'menunggu' || z.status === 'disetujui') && !(z.tanggal_selesai < mulai || z.tanggal_mulai > selesai);
  })[0];

  const sudah = absFor_(mulai).filter(function (a) { return a.id_pengguna === uid && a.tanggal >= mulai && a.tanggal <= selesai; })[0];

  if (d.lampiran) {

}

function apiAdminDashboard_(ctx) {
  const now = todayJ_(), set = getSet_();
  const dari30 = addDays_(now.tgl, -29), awalBulan = now.tgl.slice(0, 7) + '-01';
  const dari = dari30 < awalBulan ? dari30 : awalBulan;
  const idx = buildIndex_(dari, now.tgl);
  const guru = guruAktif_();
  const st = { dijadwalkan: 0, sudah: 0, tepat: 0, terlambat: 0, luar: 0, izinSakit: 0, izin: 0, sakit: 0, belum: 0, guru: guru.length };
  const terlambat = [], belum = [], hadir = [], izinHariIni = [];
  const nowM = menit_(now.jam);
  guru.forEach(function (u) {
    const s = statusHari_(idx, u.id_pengguna, now.tgl);
    if (s.jadwal.length) st.dijadwalkan++;
    if (s.absen) {
      st.sudah++;
      const tk = (tkbById_(s.absen.id_tkb) || {}).nama_tkb || s.absen.id_tkb;
      const item = { id: u.id_pengguna, nama: u.nama_lengkap, tkb: tk, jam: s.absen.jam_absen, menit: num_(s.absen.menit_terlambat, 0), status: s.absen.status, id_absen: s.absen.id_absen };
      hadir.push(item);
      if (s.kode === 'H') st.tepat++;
      if (s.kode === 'T') { st.terlambat++; terlambat.push(item); }
      if (s.kode === 'L') st.luar++;
    } else if (s.kode === 'I' || s.kode === 'S') {
      st.izinSakit++; st[s.kode === 'S' ? 'sakit' : 'izin']++;
      izinHariIni.push({ id: u.id_pengguna, nama: u.nama_lengkap, jenis: s.izin.jenis });
    } else if (s.kode === 'B' || s.kode === 'A') {
      st.belum++;
      const next = s.jadwal.filter(function (j) { return menit_(j.jam_selesai) >= nowM; })[0] || s.jadwal[0];
      belum.push({ id: u.id_pengguna, nama: u.nama_lengkap, hp: u.no_hp || '', mulai: next.jam_mulai, selesai: next.jam_selesai,
        mapel: next.mata_pelajaran, tkb: (tkbById_(next.id_tkb) || {}).nama_tkb || next.id_tkb, lewat: menit_(next.jam_mulai) < nowM });
    }
  });
  terlambat.sort(function (a, b) { return b.menit - a.menit; });
  belum.sort(function (a, b) { return menit_(a.mulai) - menit_(b.mulai); });

  // Grafik 30 hari
  const tren = dateRange_(dari30, now.tgl).map(function (t) {
    let h = 0, j = 0, iz = 0;
    guru.forEach(function (u) {
      const s = statusHari_(idx, u.id_pengguna, t);
      if (s.jadwal.length) j++;
      if (s.absen) h++;
      if (s.kode === 'I' || s.kode === 'S') iz++;
    });
    return { tanggal: t, hadir: h, dijadwalkan: j, izin: iz };
  });
  // Komposisi bulan berjalan
  const komp = { tepat: 0, terlambat: 0, luar: 0, izin: 0, sakit: 0, alpa: 0 };
  const datesBulan = dateRange_(awalBulan, now.tgl);
  guru.forEach(function (u) {
    const r = ringkas_(idx, u.id_pengguna, datesBulan, 0);
    komp.tepat += r.H; komp.terlambat += r.T; komp.luar += r.L; komp.izin += r.I; komp.sakit += r.S; komp.alpa += r.A;
  });
  // Persetujuan
  const izinMenunggu = izinRows_().filter(function (z) { return z.status === 'menunggu'; })
    .map(function (z) { const o = izinPublik_(z); o.nama = namaUser_(z.id_pengguna); return o; })
    .sort(function (a, b) { return a.diajukan < b.diajukan ? -1 : 1; });
  const akunMenunggu = users_().filter(function (u) { return u.google_status === 'menunggu'; })
    .map(function (u) { return { id: u.id_pengguna, nama: u.nama_lengkap, email: u.email_google, baru: u.status_akun === 'menunggu', dibuat: u.dibuat_pada }; });

  return ok_({
    tanggal: now.tgl, hari: now.hari, jam: now.detik,
    statistik: st, terlambat: terlambat, belum: belum, hadir: hadir, izinHariIni: izinHariIni,
    tren: tren, komposisi: komp,
    persetujuan: { izin: izinMenunggu, akun: akunMenunggu, total: izinMenunggu.length + akunMenunggu.length },
    tkbAktif: tkbs_().filter(function (t) { return t.status === 'aktif'; }).length,
    set: publicSet_()
  });
}

function apiAbsensiList_(ctx, d) {
  const now = todayJ_();
  const dari = validTgl_(d.dari) ? d.dari : now.tgl.slice(0, 7) + '-01';
  const sampai = validTgl_(d.sampai) ? d.sampai : now.tgl;
  const rows = absFor_(dari).filter(function (a) {
    if (a.tanggal < dari || a.tanggal > sampai) return false;
    if (d.id_tkb && a.id_tkb !== d.id_tkb) return false;
    if (d.id_pengguna && a.id_pengguna !== d.id_pengguna) return false;
    if (d.status === 'perlu_tinjau') return !isYa_(a.verifikasi) || isYa_(a.ditandai);
    if (d.status && a.status !== d.status) return false;
    return true;
  }).map(function (a) {
    const o = absPublik_(a); const u = userById_(a.id_pengguna);
    o.id_pengguna = a.id_pengguna; o.nama = u ? u.nama_lengkap : a.id_pengguna; o.nip = u ? u.nip_nuptk : '';
    return o;
  }).sort(function (a, b) { return a.tanggal < b.tanggal ? 1 : a.tanggal > b.tanggal ? -1 : (a.jam < b.jam ? 1 : -1); });
  return ok_({ dari: dari, sampai: sampai, rows: rows.slice(0, 3000), total: rows.length });
}

function apiAbsensiSimpan_(ctx, d) {
  const alasan = String(d.alasan || '').trim();
  if (alasan.length < 5) return fail_('Alasan koreksi wajib diisi (minimal 5 karakter).');
  const status = String(d.status || '');
  if (['tepat_waktu', 'terlambat', 'di_luar_jadwal'].indexOf(status) < 0) return fail_('Status tidak valid.');
  if (!validTgl_(d.tanggal)) return fail_('Tanggal tidak valid.');
  if (d.tanggal > todayJ_().tgl) return fail_('Tidak bisa mencatat absen untuk tanggal yang akan datang.');
  const jam = String(d.jam || '').slice(0, 5);
  if (!validJam_(jam)) return fail_('Jam tidak valid (format JJ:MM).');
  const u = userById_(d.id_pengguna); if (!u) return fail_('Guru tidak ditemukan.');
  const t = tkbById_(d.id_tkb); if (!t) return fail_('TKB tidak ditemukan.');
  const set = getSet_();
  const cocok = cocokJadwal_(u.id_pengguna, t.id_tkb, d.tanggal, jam, num_(set.toleransi_terlambat_menit, 15));
  let menit = status === 'terlambat' ? (num_(d.menit, 0) || cocok.menit || 0) : 0;
  const catatan = 'Koreksi admin (' + ctx.me.nama + ', ' + fmtD_(new Date(), 'dd/MM/yyyy HH:mm') + '): ' + alasan;
  const hasil = withLock_(function () {
    if (d.id) {
      const lama = readSheet_(SH.abs).filter(function (a) { return a.id_absen === d.id; })[0];
      if (!lama) throw new Error('Data absen tidak ditemukan.');
      if (lama.id_pengguna !== u.id_pengguna || lama.tanggal !== d.tanggal) {
        const dup = sudahAbsen_(u.id_pengguna, d.tanggal, true);
        if (dup && dup.id_absen !== d.id) throw new Error('Guru ini sudah punya data absen pada tanggal tersebut.');
      }
      const o = updateObj_(SH.abs, 'id_absen', d.id, {
        id_pengguna: u.id_pengguna, tanggal: d.tanggal, id_tkb: t.id_tkb, jam_absen: jam.length === 5 ? jam + ':00' : jam,
        status: status, menit_terlambat: menit, id_jadwal_terkait: cocok.jadwal ? cocok.jadwal.id : lama.id_jadwal_terkait,
        verifikasi: 'ya', ditandai: '', diubah_admin: 'ya', catatan_admin: catatan
      });
      CacheService.getScriptCache().remove('absday_' + lama.id_pengguna + '_' + lama.tanggal);
      return o;
    }
    if (sudahAbsen_(u.id_pengguna, d.tanggal, true)) throw new Error('Guru ini sudah punya data absen pada tanggal tersebut. Ubah data yang ada.');
    const o = { id_absen: newId_('ABS'), tanggal: d.tanggal, id_pengguna: u.id_pengguna, id_tkb: t.id_tkb, jam_absen: jam + ':00',
      latitude: '', longitude: '', akurasi_gps_meter: '', jarak_ke_tkb_meter: '', foto_file_id: '', jenis_foto: '',
      id_jadwal_terkait: cocok.jadwal ? cocok.jadwal.id : '', status: status, menit_terlambat: menit, verifikasi: 'ya',
      ditandai: '', diubah_admin: 'ya', catatan_admin: catatan, perangkat: 'manual-admin', dibuat_pada: nowIso_() };
    appendObj_(SH.abs, o);
    return o;
  });
  audit_(ctx, d.id ? 'ABSEN_KOREKSI' : 'ABSEN_TAMBAH_MANUAL', 'Absensi:' + hasil.id_absen, u.nama_lengkap + ' ' + d.tanggal + ' → ' + status + ' | ' + alasan);
  return ok_({ absen: absPublik_(hasil) }, d.id ? 'Koreksi tersimpan.' : 'Absen manual ditambahkan.');
}

function apiAbsensiVerifikasi_(ctx, d) {
  const patch = {};
  if (d.verifikasi !== undefined) patch.verifikasi = d.verifikasi ? 'ya' : 'tidak';
  if (d.ditandai !== undefined) { patch.ditandai = d.ditandai ? 'ya' : ''; if (d.ditandai) patch.verifikasi = 'tidak'; }
  if (d.catatan) patch.catatan_admin = String(d.catatan).slice(0, 300);
  if (!Object.keys(patch).length) return fail_('Tidak ada perubahan.');
  const o = withLock_(function () { return updateObj_(SH.abs, 'id_absen', d.id, patch); });
  audit_(ctx, d.ditandai ? 'ABSEN_TANDAI' : 'ABSEN_VERIFIKASI', 'Absensi:' + d.id, JSON.stringify(patch));
  return ok_({ absen: absPublik_(o) }, d.ditandai ? 'Absen ditandai mencurigakan (tidak dihitung honor).' : patch.verifikasi === 'ya' ? 'Absen diverifikasi.' : 'Verifikasi dibatalkan.');
}

function apiIzinProses_(ctx, d) {
  const kep = String(d.keputusan || '');
  if (['disetujui', 'ditolak', 'dibatalkan'].indexOf(kep) < 0) return fail_('Keputusan tidak valid.');
  const z = readSheet_(SH.izin).filter(function (x) { return x.id_izin === d.id; })[0];
  if (!z) return fail_('Pengajuan tidak ditemukan.');
  if (kep === 'dibatalkan' && z.status !== 'disetujui') return fail_('Hanya izin yang sudah disetujui yang bisa dibatalkan.');
  if (kep !== 'dibatalkan' && z.status !== 'menunggu') return fail_('Pengajuan ini sudah diproses (' + z.status + ').');
  if (kep === 'ditolak' && String(d.catatan || '').trim().length < 3) return fail_('Tuliskan catatan alasan penolakan untuk guru.');
  let peringatan = '';
  if (kep === 'disetujui') {
    const bentrok = absFor_(z.tanggal_mulai).filter(function (a) { return a.id_pengguna === z.id_pengguna && a.tanggal >= z.tanggal_mulai && a.tanggal <= z.tanggal_selesai; });
    if (bentrok.length) peringatan = ' Catatan: guru tercatat hadir pada ' + bentrok.map(function (a) { return tglIndo_(a.tanggal); }).join(', ') + ' — hari itu tetap dihitung hadir.';
  }
  withLock_(function () {
    updateObj_(SH.izin, 'id_izin', z.id_izin, { status: kep, diproses_oleh: ctx.me.nama, diproses_pada: nowIso_(), catatan_admin: String(d.catatan || '').slice(0, 300) });
  });
  audit_(ctx, 'IZIN_' + kep.toUpperCase(), 'Izin:' + z.id_izin, namaUser_(z.id_pengguna) + ' ' + z.jenis + ' ' + z.tanggal_mulai + '..' + z.tanggal_selesai);
  return ok_(null, (kep === 'disetujui' ? 'Pengajuan disetujui.' : kep === 'ditolak' ? 'Pengajuan ditolak.' : 'Izin dibatalkan.') + peringatan);
}

function apiAkunProses_(ctx, d) {
  const kep = d.keputusan === 'setujui' ? 'setujui' : d.keputusan === 'tolak' ? 'tolak' : '';
  if (!kep) return fail_('Keputusan tidak valid.');
  const res = withLock_(function () {

    if (!u || u.google_status !== 'menunggu') throw new Error('Permintaan akun tidak ditemukan atau sudah diproses.');
    if (kep === 'tolak') {
      updateObj_(SH.user, 'id_pengguna', u.id_pengguna, u.status_akun === 'menunggu'
        ? { status_akun: 'ditolak', google_status: 'ditolak', diubah_pada: nowIso_() }
        : { google_status: 'ditolak', diubah_pada: nowIso_() });
      return 'Akun Google ' + u.email_google + ' ditolak.';
    }
    if (d.tautkan_ke && d.tautkan_ke !== u.id_pengguna) {
      const target = rows.filter(function (x) { return x.id_pengguna === d.tautkan_ke; })[0];
      if (!target) throw new Error('Data guru tujuan tidak ditemukan.');
      const lain = rows.filter(function (x) { return x.id_pengguna !== u.id_pengguna && x.id_pengguna !== target.id_pengguna && String(x.email_google).toLowerCase() === String(u.email_google).toLowerCase(); })[0];
      if (lain) throw new Error('Email ini sudah dipakai ' + lain.nama_lengkap + '.');
      updateObj_(SH.user, 'id_pengguna', target.id_pengguna, { email_google: u.email_google, google_status: 'disetujui', diubah_pada: nowIso_() });
      if (u.status_akun === 'menunggu') deleteObj_(SH.user, 'id_pengguna', u.id_pengguna);
      else updateObj_(SH.user, 'id_pengguna', u.id_pengguna, { email_google: '', google_status: '' });
      return 'Akun Google ' + u.email_google + ' ditautkan ke ' + target.nama_lengkap + '.';
    }
    const peran = d.peran === 'admin' ? 'admin' : u.peran || 'guru';
    updateObj_(SH.user, 'id_pengguna', u.id_pengguna, { status_akun: 'aktif', google_status: 'disetujui', peran: peran,
      ikut_absen: u.ikut_absen || 'ya', diubah_pada: nowIso_() });

  });

  return ok_(null, res);
}

function apiGuruSimpan_(ctx, d) {
  const nama = String(d.nama || '').trim();
  const nip = String(d.nip || '').trim();
  const email = String(d.email || '').trim().toLowerCase();
  const hp = waOrNip_(String(d.hp || ''), 'no_hp');
  const peran = d.peran === 'admin' ? 'admin' : 'guru';
  const status = ['aktif', 'nonaktif'].indexOf(d.status) >= 0 ? d.status : 'aktif';
  const ikut = peran === 'guru' ? 'ya' : (d.ikut_absen ? 'ya' : 'tidak');
  if (nama.length < 3) return fail_('Nama lengkap wajib diisi.');
  if (!/^[A-Za-z0-9._-]{3,30}$/.test(nip)) return fail_('NIP/NUPTK/username 3–30 karakter (huruf, angka, titik, strip).');
  if (email && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) return fail_('Format email tidak valid.');
  if (hp && !/^0\d{8,14}$/.test(hp)) return fail_('Nomor HP tidak valid.');
  let sandiBaru = '';
  const res = withLock_(function () {
    const rows = readSheet_(SH.user);
    const dupNip = rows.filter(function (x) { return x.id_pengguna !== d.id && String(x.nip_nuptk).toLowerCase() === nip.toLowerCase(); })[0];
    if (dupNip) throw new Error('NIP/username sudah dipakai ' + dupNip.nama_lengkap + '.');
    if (email) {
      const dupE = rows.filter(function (x) { return x.id_pengguna !== d.id && String(x.email_google).toLowerCase() === email; })[0];
      if (dupE) throw new Error('Email Google sudah dipakai ' + dupE.nama_lengkap + '.');
    }
    const gStatus = email ? (d.izinkan_google === false ? 'menunggu' : 'disetujui') : '';
    if (d.id) {
      const lama = rows.filter(function (x) { return x.id_pengguna === d.id; })[0];
      if (!lama) throw new Error('Data guru tidak ditemukan.');
      if (lama.peran === 'admin' && lama.status_akun === 'aktif' && (peran !== 'admin' || status !== 'aktif') && adminAktifLain_(rows, lama.id_pengguna) === 0)
        throw new Error('Harus tersisa minimal 1 admin aktif. Angkat admin lain terlebih dahulu.');
      const patch = { nama_lengkap: nama, nip_nuptk: nip, email_google: email, mata_pelajaran: String(d.mapel || '').trim(), no_hp: hp,
        peran: peran, status_akun: status, ikut_absen: ikut, diubah_pada: nowIso_() };
      if (email !== String(lama.email_google).toLowerCase() || (email && d.izinkan_google !== undefined)) patch.google_status = gStatus;
      if (!email) patch.google_status = '';
      const o = updateObj_(SH.user, 'id_pengguna', d.id, patch);
      if (status !== 'aktif' || peran !== lama.peran) revokeAll_(d.id);
      return o;
    }
    const id = newId_('GR');
    if (d.sandi) { const v = validasiSandiBaru_(d.sandi); if (v) throw new Error(v); sandiBaru = String(d.sandi); }
    else sandiBaru = sandiAcak_(10);
    const o = { id_pengguna: id, nama_lengkap: nama, nip_nuptk: nip, email_google: email, mata_pelajaran: String(d.mapel || '').trim(), no_hp: hp,
      peran: peran, status_akun: status, ikut_absen: ikut, google_status: gStatus, wajib_ganti_sandi: 'ya', catatan: '',
      dibuat_pada: nowIso_(), diubah_pada: nowIso_() };
    appendObj_(SH.user, o);
    setSandiPengguna_(id, sandiBaru);
    return o;
  });
  audit_(ctx, d.id ? 'GURU_UBAH' : 'GURU_TAMBAH', 'Pengguna:' + res.id_pengguna, nama + ' (' + peran + ', ' + status + ')');
  const out = { guru: publicUser_(res) };
  if (!d.id) out.sandiAwal = d.sandi ? '' : sandiBaru;   // hanya ditampilkan sekali bila dibangkitkan sistem
  return ok_(out, d.id ? 'Data guru diperbarui.' : 'Guru ditambahkan.');
}

function tokenBaru_() { return 'TT' + acak_(10, 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'); }
function apiTkbSimpan_(ctx, d) {
  const nama = String(d.nama || '').trim();
  if (nama.length < 3) return fail_('Nama TKB wajib diisi.');
  const lat = d.lat === '' || d.lat === null || d.lat === undefined ? '' : Number(d.lat);
  const lng = d.lng === '' || d.lng === null || d.lng === undefined ? '' : Number(d.lng);
  if (lat !== '' && (!isFinite(lat) || lat < -90 || lat > 90)) return fail_('Latitude tidak valid (−90 s.d. 90).');
  if (lng !== '' && (!isFinite(lng) || lng < -180 || lng > 180)) return fail_('Longitude tidak valid (−180 s.d. 180).');
  const radius = Math.round(num_(d.radius, num_(getSet_().radius_gps_meter, 50)));
  if (radius < 10 || radius > 1000) return fail_('Radius antara 10 dan 1000 meter.');
  const status = d.status === 'nonaktif' ? 'nonaktif' : 'aktif';
  const res = withLock_(function () {
    const rows = readSheet_(SH.tkb);
    const dup = rows.filter(function (x) { return x.id_tkb !== d.id && String(x.nama_tkb).toLowerCase() === nama.toLowerCase(); })[0];
    if (dup) throw new Error('Nama TKB sudah ada.');
    const patch = { nama_tkb: nama, alamat: String(d.alamat || '').trim(), penanggung_jawab: String(d.pj || '').trim(),
      latitude: lat, longitude: lng, radius_meter: radius, status: status, diubah_pada: nowIso_() };
    if (d.id) return updateObj_(SH.tkb, 'id_tkb', d.id, patch);
    const no = rows.length + 1; let id = 'TKB-' + ('0' + no).slice(-2), k = no;
    while (rows.some(function (x) { return x.id_tkb === id; })) { k++; id = 'TKB-' + ('0' + k).slice(-2); }
    patch.id_tkb = id; patch.token_qr = tokenBaru_(); patch.dibuat_pada = nowIso_();
    appendObj_(SH.tkb, patch);
    return patch;
  });
  audit_(ctx, d.id ? 'TKB_UBAH' : 'TKB_TAMBAH', 'TKB:' + res.id_tkb, nama + ' r=' + radius + 'm');
  const o = tkbPublik_(res); o.token = res.token_qr;
  return ok_({ tkb: o }, d.id ? 'TKB diperbarui.' : 'TKB ditambahkan. Cetak QR-nya untuk ditempel di lokasi.');
}

}

function apiJadwalSimpan_(ctx, d) {
  if (!userById_(d.id_pengguna)) return fail_('Pilih guru.');
  if (!tkbById_(d.id_tkb)) return fail_('Pilih TKB.');
  if (HARI.indexOf(d.hari) < 0) return fail_('Pilih hari.');
  if (!validJam_(d.mulai) || !validJam_(d.selesai)) return fail_('Jam mulai/selesai tidak valid (JJ:MM).');
  if (menit_(d.selesai) <= menit_(d.mulai)) return fail_('Jam selesai harus setelah jam mulai.');
  if (String(d.mapel || '').trim().length < 2) return fail_('Mata pelajaran wajib diisi.');
  if (d.berlaku_mulai && !validTgl_(d.berlaku_mulai)) return fail_('Tanggal mulai berlaku tidak valid.');
  if (d.berlaku_sampai && !validTgl_(d.berlaku_sampai)) return fail_('Tanggal akhir berlaku tidak valid.');
  if (d.berlaku_mulai && d.berlaku_sampai && d.berlaku_sampai < d.berlaku_mulai) return fail_('Rentang berlaku tidak valid.');
  const mkBaru = daftarMingguKe_(Array.isArray(d.minggu_ke) ? d.minggu_ke.join(',') : d.minggu_ke);
  const res = withLock_(function () {
    pastikanSheet_(SH.jadwal);
    const rows = readSheet_(SH.jadwal);
    const bentrok = rows.filter(function (j) {
      if (j.id_jadwal === d.id || j.id_pengguna !== d.id_pengguna || j.hari !== d.hari) return false;
      if (j.aktif !== '' && !isYa_(j.aktif)) return false;
      const mkLama = daftarMingguKe_(j.minggu_ke);
      if (mkBaru.length && mkLama.length && !mkBaru.some(function (x) { return mkLama.indexOf(x) >= 0; })) return false;   // minggu ke berbeda → tidak bentrok
      const a1 = j.berlaku_mulai || '0000', a2 = j.berlaku_sampai || '9999', b1 = d.berlaku_mulai || '0000', b2 = d.berlaku_sampai || '9999';
      if (a2 < b1 || b2 < a1) return false;
      return menit_(d.mulai) < menit_(j.jam_selesai) && menit_(j.jam_mulai) < menit_(d.selesai);
    })[0];
    if (bentrok) throw new Error('Bentrok dengan jadwal ' + bentrok.mata_pelajaran + ' (' + bentrok.jam_mulai + '–' + bentrok.jam_selesai + ') di hari yang sama.');
    const o = { id_pengguna: d.id_pengguna, id_tkb: d.id_tkb, hari: d.hari, jam_mulai: d.mulai, jam_selesai: d.selesai,
      mata_pelajaran: String(d.mapel).trim(), kelas: String(d.kelas || '').trim(), ruang: String(d.ruang || '').trim(),
      berlaku_mulai: d.berlaku_mulai || '', berlaku_sampai: d.berlaku_sampai || '', aktif: d.aktif === false ? 'tidak' : 'ya',
      minggu_ke: mkBaru.map(function (n) { return ROMAWI[n]; }).join(',') };
    if (d.id) return updateObj_(SH.jadwal, 'id_jadwal', d.id, o);
    o.id_jadwal = newId_('JDW');
    appendObj_(SH.jadwal, o);
    return o;
  });
  audit_(ctx, d.id ? 'JADWAL_UBAH' : 'JADWAL_TAMBAH', 'Jadwal:' + res.id_jadwal, namaUser_(d.id_pengguna) + ' ' + d.hari + ' ' + d.mulai);
  return ok_({ jadwal: jadwalPublik_(res) }, 'Jadwal tersimpan.');
}

/** Tulis beberapa kunci Pengaturan sekaligus (upsert) — satu kali tulis. */
function setPengaturan_(patch) {
  withLock_(function () {
    const sh = sheet_(SH.set), v = sh.getDataRange().getValues();
    const idx = {}; for (let r = 1; r < v.length; r++) idx[String(v[r][0])] = r;
    const baru = [];
    Object.keys(patch).forEach(function (k) {
      if (idx[k] !== undefined) v[idx[k]][1] = String(patch[k]);
      else baru.push([k, String(patch[k]), '']);
    });
    sh.getRange(1, 1, v.length, 3).setValues(v.map(function (r) { return [r[0], r[1], r[2] === undefined ? '' : r[2]]; }));
    if (baru.length) sh.getRange(v.length + 1, 1, baru.length, 3).setValues(baru);
    bump_('set');
  });
}
const SET_BISA_DIUBAH = Object.keys(DEFAULT_SET).filter(function (k) { return k !== 'logo_file_id' && k !== 'logo_thumb' && k !== 'logo_kop'; });
function apiPengaturanSimpan_(ctx, d) {
  const p = d.patch || {}, out = {}, lama = getSet_();
  Object.keys(p).forEach(function (k) { if (SET_BISA_DIUBAH.indexOf(k) >= 0) out[k] = String(p[k] === null || p[k] === undefined ? '' : p[k]).trim().slice(0, 300); });
  if (out.warna_utama !== undefined && !/^#[0-9a-fA-F]{6}$/.test(out.warna_utama)) return fail_('Warna harus berformat #RRGGBB.');
  if (out.nama_aplikasi !== undefined && out.nama_aplikasi.length < 2) return fail_('Nama aplikasi minimal 2 karakter.');
  const angka = { tarif_per_kehadiran: [0, 1e8], radius_gps_meter: [10, 1000], toleransi_terlambat_menit: [0, 180], akurasi_maks_meter: [10, 1000] };
  for (const k in angka) {
    if (out[k] === undefined) continue;
    const n = Number(out[k]);
    if (!isFinite(n) || n < angka[k][0] || n > angka[k][1]) return fail_('Nilai ' + k.replace(/_/g, ' ') + ' harus antara ' + angka[k][0] + ' dan ' + angka[k][1] + '.');
    out[k] = String(Math.round(n));
  }
  if (out.kontak_admin_wa) out.kontak_admin_wa = waOrNip_(out.kontak_admin_wa, 'no_hp');
  if (!Object.keys(out).length) return fail_('Tidak ada perubahan.');
  setPengaturan_(out);
  const berubah = Object.keys(out).filter(function (k) { return String(lama[k]) !== out[k]; });
  audit_(ctx, 'PENGATURAN_UBAH', 'Pengaturan', berubah.map(function (k) { return k + ': ' + String(lama[k]).slice(0, 30) + ' → ' + out[k].slice(0, 30); }).join('; '));
  return ok_({ set: getSet_() }, 'Pengaturan tersimpan.');
}

    if (d.hapus) { setPengaturan_({ logo_kop: '' }); return ok_({ set: getSet_() }, 'Logo kop dihapus.'); }
    const t = String(d.thumb || '');
    if (!/^data:image\/(png|jpeg|webp);base64,/.test(t)) return fail_('Logo kop tidak valid.');
    if (t.length > 48000) return fail_('Logo kop terlalu besar. Gunakan gambar yang lebih sederhana.');
    setPengaturan_({ logo_kop: t });
    audit_(ctx, 'LOGO_KOP_UBAH', 'Pengaturan', 'logo kop diperbarui');
    return ok_({ set: getSet_() }, 'Logo kop surat diperbarui.');
  }
  if (d.hapus) {
    setPengaturan_({ logo_file_id: '', logo_thumb: '' });
    audit_(ctx, 'LOGO_HAPUS', 'Pengaturan', '');
    return ok_({ set: getSet_() }, 'Logo dihapus. Ikon bawaan dipakai kembali.');
  }
  const thumb = String(d.thumb || '');
  if (!/^data:image\/(png|jpeg|webp);base64,/.test(thumb)) return fail_('Thumbnail logo tidak valid.');
  if (thumb.length > 48000) return fail_('Thumbnail terlalu besar. Gunakan gambar yang lebih sederhana.');
  let fileId = '';
  if (d.asli) {
    const f = simpanDataUrl_(d.asli, folderProp_('FOLDER_IDENTITAS', 'DRIVE_ROOT_FOLDER_ID', 'Identitas Aplikasi'), 'logo_' + fmtD_(new Date(), 'yyyyMMdd_HHmmss') + '.png', 2.5e6);
    fileId = f.getId();
  }
  setPengaturan_({ logo_file_id: fileId, logo_thumb: thumb });
  audit_(ctx, 'LOGO_UBAH', 'Pengaturan', 'logo diperbarui');
  return ok_({ set: getSet_() }, 'Logo diperbarui.');
}

const RAHASIA_DASHBOARD = {

};
function apiRahasiaSimpan_(ctx, d) {
  const k = String(d.kunci || ''), aturan = RAHASIA_DASHBOARD[k];

  audit_(ctx, 'RAHASIA_UBAH', 'Rahasia:' + k, d.nilai === '__HAPUS__' ? 'dihapus' : 'diperbarui');

/**
 * ============================================================================
 *  TeachTrack — Modul_Laporan.gs
 *  Rekap absensi & laporan honorarium untuk rentang tanggal bebas,
 *  filter hari / guru / TKB, grafik ringkas, dan PDF siap cetak.
 *  Rumus honor: Jumlah kehadiran TERVERIFIKASI × tarif per kehadiran.
 * ============================================================================
 */

function hitungRekap_(d) {
  const now = todayJ_(), set = getSet_();
  const dari = validTgl_(d.dari) ? d.dari : now.tgl.slice(0, 7) + '-01';
  let sampai = validTgl_(d.sampai) ? d.sampai : now.tgl;
  if (sampai < dari) throw new Error('Tanggal akhir tidak boleh sebelum tanggal awal.');
  if (dateRange_(dari, sampai).length > 400) throw new Error('Rentang maksimal ±13 bulan per laporan.');
  const efektif = sampai > now.tgl ? now.tgl : sampai;
  const hariFilter = Array.isArray(d.hari) ? d.hari.filter(function (h) { return HARI.indexOf(h) >= 0; }) : [];
  const tarif = num_(set.tarif_per_kehadiran, 0);
  const tkbF = d.id_tkb && tkbById_(d.id_tkb) ? d.id_tkb : '';
  const idx = buildIndex_(dari, efektif < dari ? dari : efektif, { tkb: tkbF });
  const dates = efektif < dari ? [] : dateRange_(dari, efektif).filter(function (t) { return !hariFilter.length || hariFilter.indexOf(hariOf_(t)) >= 0; });

  let calon = users_().filter(function (u) { return ikutAbsen_(u) && u.status_akun !== 'menunggu' && u.status_akun !== 'ditolak'; });
  if (d.id_pengguna) calon = calon.filter(function (u) { return u.id_pengguna === d.id_pengguna; });

  const rows = [], perBulan = {}, perTkb = {}, rincian = [];
  const komp = { tepat: 0, terlambat: 0, luar: 0, izin: 0, sakit: 0, alpa: 0 };
  calon.forEach(function (u) {
    const uid = u.id_pengguna;
    const r = { id: uid, nama: u.nama_lengkap, nip: u.nip_nuptk, mapel: u.mata_pelajaran || '', status_akun: u.status_akun,
      hadir: 0, tepat: 0, terlambat: 0, luar: 0, terverifikasi: 0, izin: 0, sakit: 0, alpa: 0, menitTerlambat: 0, hariJadwal: 0, perTkbHonor: {} };
    dates.forEach(function (t) {
      const s = statusHari_(idx, uid, t);
      if (s.jadwal.length) r.hariJadwal++;
      const bln = t.slice(0, 7);
      if (!perBulan[bln]) perBulan[bln] = { bulan: bln, hadir: 0, terlambat: 0, izin: 0, sakit: 0, alpa: 0 };
      const pb = perBulan[bln];
      let ket = '', tkbNama = '', jam = '';
      if (s.absen) {
        r.hadir++; pb.hadir++;
        tkbNama = (tkbById_(s.absen.id_tkb) || {}).nama_tkb || s.absen.id_tkb; jam = String(s.absen.jam_absen || '').slice(0, 5);
        if (!perTkb[tkbNama]) perTkb[tkbNama] = { tkb: tkbNama, hadir: 0, terlambat: 0 };
        perTkb[tkbNama].hadir++;
        if (isYa_(s.absen.verifikasi)) { r.terverifikasi++; r.perTkbHonor[s.absen.id_tkb] = (r.perTkbHonor[s.absen.id_tkb] || 0) + 1; }
        if (s.kode === 'H') { r.tepat++; komp.tepat++; }
        if (s.kode === 'T') { r.terlambat++; komp.terlambat++; pb.terlambat++; perTkb[tkbNama].terlambat++; r.menitTerlambat += num_(s.absen.menit_terlambat, 0); ket = 'Terlambat ' + num_(s.absen.menit_terlambat, 0) + ' mnt'; }
        if (s.kode === 'L') { r.luar++; komp.luar++; ket = 'Di luar jadwal' + (isYa_(s.absen.verifikasi) ? '' : ' (belum diverifikasi)'); }
        if (isYa_(s.absen.ditandai)) ket = 'Ditandai mencurigakan';
      } else if ((s.kode === 'I' || s.kode === 'S') && (!tkbF || s.jadwal.length)) {
        if (s.kode === 'S') { r.sakit++; komp.sakit++; pb.sakit++; } else { r.izin++; komp.izin++; pb.izin++; }
        ket = s.izin.alasan ? String(s.izin.alasan).slice(0, 60) : '';
      } else if (s.kode === 'A') { r.alpa++; komp.alpa++; pb.alpa++; }
      else return;
      if (s.kode === 'B') return;
      rincian.push({ tanggal: t, hari: hariOf_(t), id: uid, nama: u.nama_lengkap, kode: s.kode, tkb: tkbNama, jam: jam,
        verifikasi: s.absen ? isYa_(s.absen.verifikasi) : null, ket: ket,
        jadwal: s.jadwal.map(function (j) { return j.jam_mulai + ' ' + j.mata_pelajaran; }).join(', ') });
    });
    r.honor = r.terverifikasi * tarif;
    const dasar = r.hadir + r.alpa;
    r.persen = dasar ? Math.round(r.hadir / dasar * 1000) / 10 : null;
    if (tkbF && !r.hariJadwal && !r.hadir) return;               // guru tanpa kaitan dengan TKB terpilih
    if (u.status_akun !== 'aktif' && !r.hadir && !r.izin && !r.sakit) return;
    rows.push(r);
  });
  rows.sort(function (a, b) { return a.nama.localeCompare(b.nama); });
  const total = { hadir: 0, tepat: 0, terlambat: 0, luar: 0, terverifikasi: 0, izin: 0, sakit: 0, alpa: 0, honor: 0, guru: rows.length };
  rows.forEach(function (r) { Object.keys(total).forEach(function (k) { if (k !== 'guru') total[k] += r[k]; }); });
  rincian.sort(function (a, b) { return a.tanggal < b.tanggal ? -1 : a.tanggal > b.tanggal ? 1 : a.nama.localeCompare(b.nama); });
  return {
    dari: dari, sampai: sampai, efektifSampai: efektif, hari: hariFilter, tarif: tarif,
    filter: { id_pengguna: d.id_pengguna || '', id_tkb: tkbF, tkb: tkbF ? tkbById_(tkbF).nama_tkb : '', guru: d.id_pengguna ? namaUser_(d.id_pengguna) : '' },
    rows: rows, total: total, komposisi: komp,
    perBulan: Object.keys(perBulan).sort().map(function (k) { return perBulan[k]; }),
    perTkb: Object.keys(perTkb).map(function (k) { return perTkb[k]; }).sort(function (a, b) { return b.hadir - a.hadir; }),
    topTerlambat: rows.filter(function (r) { return r.terlambat > 0; }).sort(function (a, b) { return b.terlambat - a.terlambat || b.menitTerlambat - a.menitTerlambat; }).slice(0, 8)
      .map(function (r) { return { nama: r.nama, kali: r.terlambat, menit: r.menitTerlambat }; }),
    rincian: rincian.slice(0, 6000),
    terbilangTotal: terbilang_(total.honor)
  };
}

    const folder = jenis === 'rekap' ? folderProp_('FOLDER_LAP_REKAP', 'FOLDER_LAPORAN', 'Rekap Absensi') : folderProp_('FOLDER_LAP_HONOR', 'FOLDER_LAPORAN', 'Honorarium');

const URUTAN_IMPOR = ['Pengguna', 'TKB', 'Jadwal', 'Agenda', 'Izin', 'Absensi'];   // master dulu, relasi kemudian

function openSource_(idOrUrl) {
  const m = String(idOrUrl || '').match(/\/d\/([A-Za-z0-9_-]{20,})/);
  const id = m ? m[1] : String(idOrUrl || '').trim();
  if (!id) throw new Error('Tempel URL atau ID spreadsheet lama.');
  if (id === CONFIG.SPREADSHEET_ID) throw new Error('Itu spreadsheet aplikasi ini. Tempel spreadsheet LAMA.');
  try { return SpreadsheetApp.openById(id); }
  catch (e) { throw new Error('Spreadsheet tidak bisa dibuka. Pastikan ID benar & dimiliki/dibagikan ke akun pemilik aplikasi ini.'); }
}
function readSource_(ss, name) {
  const sh = ss.getSheetByName(name); if (!sh) return null;
  const v = sh.getDataRange().getValues(); if (v.length <= 1) return [];
  const h = v[0].map(function (x) { return String(x).trim(); });
  return v.slice(1).filter(function (r) { return r.join('') !== ''; }).map(function (r) {
    const o = {}; h.forEach(function (k, i) { if (k) o[k] = normCell_(r[i], k); }); return o;
  });
}

/**
 * plan = { target, key(o), rows[], updatable[]|'semua' }
 * Target dibaca 1×, digabung di memori, ditulis 1×.
 */
function upsertPlan_(plan, dryRun) {
  const sh = sheet_(plan.target), values = sh.getDataRange().getValues(), h = values[0].map(String);
  const index = {};
  for (let r = 1; r < values.length; r++) {
    const o = {}; h.forEach(function (k, i) { o[k] = normCell_(values[r][i], k); });
    const k = plan.key(o); if (k) index[k] = r;
  }
  const res = { sumber: plan.rows.length, ditambah: 0, diperbarui: 0, dilewati: 0 }, baru = [], seen = {};
  const upd = plan.updatable === 'semua' ? h.filter(function (f) { return f !== h[0]; }) : (plan.updatable || []);
  plan.rows.forEach(function (src) {
    const k = plan.key(src);
    if (!k || seen[k]) { res.dilewati++; return; } seen[k] = true;
    if (index[k] !== undefined) {
      const row = values[index[k]]; let changed = false;
      upd.forEach(function (f) {
        const i = h.indexOf(f); if (i < 0 || src[f] === undefined || src[f] === '') return;
        if (String(normCell_(row[i], f)) !== String(src[f])) { row[i] = src[f]; changed = true; }
      });
      changed ? res.diperbarui++ : res.dilewati++;
    } else { baru.push(h.map(function (f) { return src[f] !== undefined ? src[f] : ''; })); res.ditambah++; }
  });
  if (!dryRun && (res.diperbarui || baru.length)) {
    const all = values.concat(baru);
    sh.getRange(1, 1, all.length, h.length).setValues(all);
    if (SCOPE_OF[plan.target]) bump_(SCOPE_OF[plan.target]);
  }
  return res;
}

function apiImpor_(ctx, d, dryRun) {
  const src = openSource_(d.sumber);
  const pilih = (Array.isArray(d.sheets) && d.sheets.length ? d.sheets : URUTAN_IMPOR).filter(function (s) { return URUTAN_IMPOR.indexOf(s) >= 0; });
  const timpa = !!d.timpa;
  const report = {}, warnings = [], akunBaru = [];

  const run = function () {
    // Peta ID sumber → ID target (agar relasi tetap benar bila NIP/nama sama tetapi ID beda)
    const tgtUsers = readSheet_(SH.user), tgtTkb = readSheet_(SH.tkb);
    const srcUsers = readSource_(src, 'Pengguna') || [], srcTkb = readSource_(src, 'TKB') || [];
    const mapU = {}, mapT = {};
    srcUsers.forEach(function (u) {
      const nip = String(u.nip_nuptk || '').toLowerCase(), em = String(u.email_google || '').toLowerCase();
      const t = tgtUsers.filter(function (x) { return (nip && String(x.nip_nuptk).toLowerCase() === nip) || (em && String(x.email_google).toLowerCase() === em); })[0];
      mapU[u.id_pengguna] = t ? t.id_pengguna : u.id_pengguna;
    });
    srcTkb.forEach(function (t) {
      const x = tgtTkb.filter(function (y) { return y.id_tkb === t.id_tkb || String(y.nama_tkb).toLowerCase() === String(t.nama_tkb).toLowerCase(); })[0];
      mapT[t.id_tkb] = x ? x.id_tkb : t.id_tkb;
    });
    const uOk = function (id) { return !!mapU[id] || tgtUsers.some(function (x) { return x.id_pengguna === id; }); };
    const tOk = function (id) { return !!mapT[id] || tgtTkb.some(function (x) { return x.id_tkb === id; }); };

    pilih.sort(function (a, b) { return URUTAN_IMPOR.indexOf(a) - URUTAN_IMPOR.indexOf(b); }).forEach(function (name) {
      let rows = readSource_(src, name);
      if (rows === null) { report[name] = { sumber: 0, ditambah: 0, diperbarui: 0, dilewati: 0, catatan: 'sheet tidak ada' }; return; }
      let plan;
      if (name === 'Pengguna') {
        const nipSeen = {};
        rows = rows.filter(function (u) {
          if (u.status_akun === 'menunggu' || u.status_akun === 'ditolak') return false;
          const k = String(u.nip_nuptk || '').toLowerCase();
          if (!k) { warnings.push('Pengguna "' + u.nama_lengkap + '" tanpa NIP/username dilewati.'); return false; }
          if (nipSeen[k]) { warnings.push('NIP ganda di sumber: ' + u.nip_nuptk); return false; }
          nipSeen[k] = 1; return true;
        }).map(function (u) {
          const o = {}; HEADERS.Pengguna.forEach(function (f) { o[f] = u[f] === undefined ? '' : u[f]; });
          o.id_pengguna = mapU[u.id_pengguna] || u.id_pengguna;
          o.no_hp = waOrNip_(o.no_hp, 'no_hp'); o.wajib_ganti_sandi = 'ya';
          if (u.peran === 'admin') warnings.push('Admin lama "' + u.nama_lengkap + '" ikut diimpor sebagai admin.');
          return o;
        });
        plan = { target: SH.user, rows: rows, key: function (o) { return String(o.nip_nuptk || '').toLowerCase(); },
          updatable: timpa ? ['nama_lengkap', 'email_google', 'mata_pelajaran', 'no_hp', 'ikut_absen'] : [] };
      } else if (name === 'TKB') {
        rows = rows.map(function (t) { const o = {}; HEADERS.TKB.forEach(function (f) { o[f] = t[f] === undefined ? '' : t[f]; }); o.id_tkb = mapT[t.id_tkb] || t.id_tkb; return o; });
        plan = { target: SH.tkb, rows: rows, key: function (o) { return o.id_tkb; },
          updatable: timpa ? ['nama_tkb', 'alamat', 'penanggung_jawab', 'latitude', 'longitude', 'radius_meter'] : [] };
      } else {
        const fk = function (o, label) {
          if (o.id_pengguna !== undefined) {
            const id = mapU[o.id_pengguna] || o.id_pengguna;
            if (!uOk(o.id_pengguna)) { warnings.push(label + ': guru ' + o.id_pengguna + ' tidak ada (dilewati).'); return null; }
            o.id_pengguna = id;
          }
          if (o.id_tkb) { if (!tOk(o.id_tkb)) { warnings.push(label + ': TKB ' + o.id_tkb + ' tidak ada (dilewati).'); return null; } o.id_tkb = mapT[o.id_tkb] || o.id_tkb; }
          return o;
        };
        rows = rows.map(function (r) { const o = {}; HEADERS[name].forEach(function (f) { o[f] = r[f] === undefined ? '' : r[f]; }); return fk(o, name); }).filter(Boolean);
        const keyF = { Jadwal: function (o) { return o.id_jadwal; }, Agenda: function (o) { return o.id_agenda; },
          Izin: function (o) { return o.id_izin; }, Absensi: function (o) { return o.id_pengguna && o.tanggal ? o.id_pengguna + '|' + o.tanggal : ''; } }[name];
        const upd = { Jadwal: timpa ? 'semua' : [], Agenda: timpa ? 'semua' : [], Absensi: timpa ? 'semua' : [],
          Izin: ['status', 'diproses_oleh', 'diproses_pada', 'catatan_admin'] }[name];
        plan = { target: name, rows: rows, key: keyF, updatable: upd };
        if (name === 'Izin' && !timpa) {
          // Delta sync: status izin hanya diperbarui bila di app baru masih "menunggu"
          const tgt = {}; readSheet_(SH.izin).forEach(function (z) { tgt[z.id_izin] = z.status; });
          plan.rows = rows.filter(function (z) { return !tgt[z.id_izin] || tgt[z.id_izin] === 'menunggu'; });
          plan.rows.__skip = rows.length - plan.rows.length;
        }
      }
      const res = upsertPlan_(plan, dryRun);
      if (plan.rows.__skip) { res.dilewati += plan.rows.__skip; res.sumber += plan.rows.__skip; }
      report[name] = res;
      if (name === 'Pengguna' && !dryRun) {
        // Sandi tidak ikut pindah (tersimpan di Script Properties proyek lama) → buat sandi awal baru
        readSheet_(SH.user).forEach(function (u) {
          if (!prop_('PW_' + u.id_pengguna) && u.status_akun === 'aktif') {
            const s = sandiAcak_(10); setSandiPengguna_(u.id_pengguna, s);
            akunBaru.push({ nama: u.nama_lengkap, username: u.nip_nuptk, sandi: s });
          }
        });
      }
    });
  };
  if (dryRun) run(); else withLock_(run);
  if (!dryRun) {
    ['set', 'user', 'tkb', 'jadwal', 'agenda', 'izin', 'abs'].forEach(bump_);
    audit_(ctx, 'IMPOR_JALAN', 'Impor', src.getName() + ' | ' + JSON.stringify(report));
  }
  return ok_({ sumber: { nama: src.getName(), id: src.getId() }, dryRun: dryRun, report: report, warnings: warnings.slice(0, 50), akunBaru: akunBaru },
    dryRun ? 'Pindai selesai — belum ada data yang ditulis.' : 'Impor selesai.');
}

/** Impor guru dari CSV (diparse di browser): [{nama, nip, mapel, hp, email}] */
function apiGuruImporCsv_(ctx, d) {
  const rows = (d.rows || []).slice(0, 500);
  if (!rows.length) return fail_('Tidak ada baris untuk diimpor.');
  const hasil = { ditambah: 0, diperbarui: 0, dilewati: 0 }, akun = [], warnings = [];
  withLock_(function () {
    const sh = sheet_(SH.user), values = sh.getDataRange().getValues(), h = values[0].map(String);
    const iNip = h.indexOf('nip_nuptk'), iEmail = h.indexOf('email_google');
    const byNip = {}, byEmail = {};
    for (let r = 1; r < values.length; r++) {
      byNip[String(values[r][iNip]).toLowerCase()] = r;
      if (values[r][iEmail]) byEmail[String(values[r][iEmail]).toLowerCase()] = r;
    }
    const baru = [], seen = {};
    rows.forEach(function (x, n) {
      const nama = String(x.nama || '').trim(), nip = String(x.nip || '').trim().replace(/\s+/g, '');
      const email = String(x.email || '').trim().toLowerCase(), hp = waOrNip_(String(x.hp || ''), 'no_hp');
      if (nama.length < 3 || !/^[A-Za-z0-9._-]{3,30}$/.test(nip)) { hasil.dilewati++; warnings.push('Baris ' + (n + 2) + ': nama/NIP tidak valid.'); return; }
      const k = nip.toLowerCase();
      if (seen[k]) { hasil.dilewati++; warnings.push('Baris ' + (n + 2) + ': NIP ganda di berkas.'); return; } seen[k] = 1;
      if (email && byEmail[email] !== undefined && byEmail[email] !== byNip[k]) { hasil.dilewati++; warnings.push('Baris ' + (n + 2) + ': email sudah dipakai guru lain.'); return; }
      if (byNip[k] !== undefined) {
        const row = values[byNip[k]];
        const set = function (f, v) { const i = h.indexOf(f); if (i >= 0 && v && String(row[i]) !== v) { row[i] = v; return true; } return false; };
        const ch = [set('nama_lengkap', nama), set('mata_pelajaran', String(x.mapel || '').trim()), set('no_hp', hp), set('email_google', email)].some(Boolean);
        if (ch) { if (email) row[h.indexOf('google_status')] = row[h.indexOf('google_status')] || 'disetujui'; row[h.indexOf('diubah_pada')] = nowIso_(); hasil.diperbarui++; }
        else hasil.dilewati++;
        return;
      }
      const id = newId_('GR');
      const o = { id_pengguna: id, nama_lengkap: nama, nip_nuptk: nip, email_google: email, mata_pelajaran: String(x.mapel || '').trim(), no_hp: hp,
        peran: 'guru', status_akun: 'aktif', ikut_absen: 'ya', google_status: email ? 'disetujui' : '', wajib_ganti_sandi: 'ya',
        dibuat_pada: nowIso_(), diubah_pada: nowIso_() };
      baru.push(toRow_(h, o));
      const s = sandiAcak_(10); setSandiPengguna_(id, s);
      akun.push({ nama: nama, username: nip, sandi: s });
      hasil.ditambah++;
    });
    const all = values.concat(baru);
    sh.getRange(1, 1, all.length, h.length).setValues(all);
    bump_('user');
  });
  audit_(ctx, 'GURU_IMPOR_CSV', 'Pengguna', JSON.stringify(hasil));
  return ok_({ hasil: hasil, akunBaru: akun, warnings: warnings.slice(0, 50) }, 'Impor guru selesai: ' + hasil.ditambah + ' ditambah, ' + hasil.diperbarui + ' diperbarui.');
}

/** "JULI-AGUSTUS-SEPTEMBER TAHUN 2025" atau "DESEMBER 2025 - FEBRUARI 2026" */
function periodeBulan_(dari, sampai) {
  const out = []; let y = +dari.slice(0, 4), m = +dari.slice(5, 7); const y2 = +sampai.slice(0, 4), m2 = +sampai.slice(5, 7);
  let g = 0; while ((y < y2 || (y === y2 && m <= m2)) && g++ < 24) { out.push({ b: BULAN[m - 1].toUpperCase(), y: y }); m++; if (m > 12) { m = 1; y++; } }
  if (!out.length) return '';
  if (out[0].y === out[out.length - 1].y) return out.map(function (x) { return x.b; }).join('-') + ' TAHUN ' + out[0].y;
  return out[0].b + ' ' + out[0].y + ' - ' + out[out.length - 1].b + ' ' + out[out.length - 1].y;
}

function targetJurnal_(ctx, d) {
  const admin = ctx.me.peran === 'admin';
  let uid = ctx.me.id;
  if (d.id_pengguna && d.id_pengguna !== uid) { if (!admin) throw new Error('Anda hanya dapat membuka jurnal milik sendiri.'); uid = d.id_pengguna; }
  const u = userById_(uid); if (!u) throw new Error('Pilih guru terlebih dahulu.');

  let tkb = d.id_tkb || (milik[0] && milik[0].id) || '';
  if (!tkb) throw new Error('Belum ada TKB yang terkait dengan ' + u.nama_lengkap + ' (jadwal/absen).');
  if (!admin && !milik.some(function (t) { return t.id === tkb; })) throw new Error('TKB ini tidak terkait dengan jadwal/absen Anda.');
  if (!tkbById_(tkb)) throw new Error('TKB tidak ditemukan.');
  return { uid: uid, u: u, tkb: tkb, milik: milik };
}
function rentangBulan_(d) {
  const now = todayJ_().tgl;
  const b1 = /^\d{4}-\d{2}$/.test(String(d.dari || '')) ? d.dari : now.slice(0, 7);
  const b2 = /^\d{4}-\d{2}$/.test(String(d.sampai || '')) ? d.sampai : b1;
  if (b2 < b1) throw new Error('Bulan akhir tidak boleh sebelum bulan awal.');
  const dari = b1 + '-01', sampai = addDays_(addDays_(b2 + '-01', 32).slice(0, 7) + '-01', -1);
  if (dateRange_(dari, sampai).length > 370) throw new Error('Rentang jurnal maksimal 12 bulan.');
  return { b1: b1, b2: b2, dari: dari, sampai: sampai };
}

/** Label "3-4 Januari 2026" / "31 Januari - 1 Februari 2026" / lintas tahun. */
function labelSabtuMinggu_(sab, min) {
  const a = sab.split('-'), b = min.split('-');
  if (a[0] !== b[0]) return (+a[2]) + ' ' + BULAN[+a[1] - 1] + ' ' + a[0] + ' - ' + (+b[2]) + ' ' + BULAN[+b[1] - 1] + ' ' + b[0];
  if (a[1] !== b[1]) return (+a[2]) + ' ' + BULAN[+a[1] - 1] + ' - ' + (+b[2]) + ' ' + BULAN[+b[1] - 1] + ' ' + b[0];
  return (+a[2]) + '-' + (+b[2]) + ' ' + BULAN[+b[1] - 1] + ' ' + b[0];
}
/** Susun data jurnal per akhir pekan (Sabtu–Minggu); minggu milik bulan hari Minggu-nya. */
function susunJurnal_(t, r) {
  const abs = absFor_(addDays_(r.dari, -1)).filter(function (a) { return a.id_tkb === t.tkb && a.tanggal >= addDays_(r.dari, -1) && a.tanggal <= r.sampai; });
  const jadMap = {}; jadwalRows_().forEach(function (j) { jadMap[j.id_jadwal] = j; });
  const cat = {}; jurnalRows_().forEach(function (x) { if (x.id_pengguna === t.uid && x.id_tkb === t.tkb) cat[x.minggu] = x.catatan; });
  let min = r.dari; while (hariOf_(min) !== 'Minggu') min = addDays_(min, 1);
  const out = [];
  for (; min <= r.sampai; min = addDays_(min, 7)) {
    const sab = addDays_(min, -1);
    const pek = abs.filter(function (a) { return a.tanggal === sab || a.tanggal === min; })
      .sort(function (a, b) { return (a.tanggal + a.jam_absen) < (b.tanggal + b.jam_absen) ? -1 : 1; });
    const mapel = [], guru = [];
    pek.forEach(function (a) {
      const j = jadMap[a.id_jadwal_terkait], u = userById_(a.id_pengguna);
      const mp = (j && j.mata_pelajaran) || (u && u.mata_pelajaran) || '';
      if (mp && mapel.indexOf(mp) < 0) mapel.push(mp);
      const nm = u ? u.nama_lengkap : a.id_pengguna;
      if (guru.indexOf(nm) < 0) guru.push(nm);
    });
    const fotoA = pek.filter(function (a) { return a.foto_file_id && a.id_pengguna === t.uid; })[0] || pek.filter(function (a) { return a.foto_file_id; })[0];
    const catatan = cat[min] || '';
    if (!pek.length && !catatan) continue;
    out.push({ minggu: min, sabtu: sab, label: labelSabtuMinggu_(sab, min), ke: URUTAN_KE[mingguKe_(min)] || '', bulan: min.slice(0, 7),
      mapel: mapel, guru: guru, jumlahAbsen: pek.length, foto_id: fotoA ? fotoA.id_absen : '', foto_file: fotoA ? fotoA.foto_file_id : '', catatan: catatan });
  }
  return out;
}

function kelasKolom_(k) {
  const out = [];
  String(k || '').toUpperCase().split(/[,\/&;]+|\sDAN\s/).forEach(function (x) {
    x = x.trim();
    const v = /^(XII|12)\b/.test(x) ? 'XII' : /^(XI|11)\b/.test(x) ? 'XI' : /^(X|10)\b/.test(x) ? 'X' : '';
    if (v && out.indexOf(v) < 0) out.push(v);
  });
  return out;
}

function warmCaches_(scope) {
  const c = CacheService.getScriptCache(), flag = 'warm_' + scope;
  if (scope !== 'setup' && c.get(flag)) return 'skip';
  c.put(flag, '1', scope === 'tick' ? 50 : 60);
  getSet_(); users_(); tkbs_(); jadwalRows_(); agendaRows_(); izinRows_(); absRecent_();
  c.put('warm_last', JSON.stringify({ at: nowIso_(), scope: scope }), 21600);
  return 'ok';
}
/** Trigger tiap 1 menit — hanya bekerja pukul 05.00–19.59 WIB agar hemat kuota. */
function keepWarm() {
  if (prop_('KEEP_WARM') === 'off') return;
  const jam = Number(fmtD_(new Date(), 'H'));
  if (jam < 5 || jam > 19) return;
  try { warmCaches_('tick'); } catch (e) { console.error('keepWarm: ' + e.message); }
}
function installTriggers_() {
  ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'keepWarm') ScriptApp.deleteTrigger(t); });
  if (prop_('KEEP_WARM') === 'off') { Logger.log('ℹ️ keepWarm dimatikan (KEEP_WARM=off).'); return; }
  ScriptApp.newTrigger('keepWarm').timeBased().everyMinutes(1).create();
  Logger.log('✅ Trigger keepWarm (tiap 1 menit, aktif 05.00–20.00 WIB) terpasang.');
}

/** Tanggal hari Minggu, n pekan sebelum hari Minggu terakhir (≤ tgl). */
function mingguSebelum_(tgl, n) {
  const idx = HARI.indexOf(hariOf_(tgl));
  return addDays_(tgl, -idx - 7 * n);
}
/** Tulis banyak baris; kolom teks diformat '@' dulu agar tanggal/jam/NIP tidak diubah Sheets. */
function tulisDemo_(name, objs) {
  if (!objs.length) return;
  const sh = sheet_(name), h = headerOf_(sh), mulai = sh.getLastRow() + 1;
  if (mulai + objs.length - 1 > sh.getMaxRows()) sh.insertRowsAfter(sh.getMaxRows(), mulai + objs.length - 1 - sh.getMaxRows() + 20);
  h.forEach(function (k, i) { if (TEXT_COLS.indexOf(k) >= 0) sh.getRange(mulai, i + 1, objs.length, 1).setNumberFormat('@'); });
  sh.getRange(mulai, 1, objs.length, h.length).setValues(objs.map(function (o) {
    return h.map(function (k) { const v = o[k]; return v === undefined || v === null ? '' : (TEXT_COLS.indexOf(k) >= 0 ? String(v) : v); });
  }));
}
/** Hapus baris yang cocok (dari bawah, per blok berurutan). Mengembalikan objek baris yang dihapus. */
function hapusBaris_(name, cocok) {
  const sh = ss_().getSheetByName(name);
  if (!sh || sh.getLastRow() < 2) return [];
  const v = sh.getDataRange().getValues(), h = v[0].map(String), kena = [], out = [];
  for (let r = 1; r < v.length; r++) {
    const o = {}; h.forEach(function (k, i) { o[k] = normCell_(v[r][i], k); });
    if (cocok(o)) { kena.push(r + 1); out.push(o); }
  }
  for (let i = kena.length - 1; i >= 0;) {
    let j = i; while (j > 0 && kena[j - 1] === kena[j] - 1) j--;
    sh.deleteRows(kena[j], i - j + 1); i = j - 1;
  }
  return out;
}

  const folder = folderProp_('FOLDER_FOTO', 'DRIVE_ROOT_FOLDER_ID', 'Foto Absensi');
  const f = folder.createFile(Utilities.newBlob(Utilities.base64Decode(jenis === 'selfie' ? DEMO_FOTO_SELFIE_B64 : DEMO_FOTO_KEGIATAN_B64),
    'image/jpeg', 'DEMO_' + jenis + '.jpg'));
  setProp_(key, f.getId());
  return f.getId();
}
